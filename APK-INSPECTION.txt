    private boolean shouldShowTvLogo() {
        Configuration c = getResources().getConfiguration();

        boolean landscape =
                c.orientation == Configuration.ORIENTATION_LANDSCAPE;

        boolean television =
                (c.uiMode & Configuration.UI_MODE_TYPE_MASK)
                == Configuration.UI_MODE_TYPE_TELEVISION;

        boolean tablet = c.smallestScreenWidthDp >= 600;

        return landscape && (television || tablet);
    }

    private void setBroadcastFullscreen(boolean fullscreen) {
        if (broadcastMiniBox == null) return;

        boolean wasFullscreen = broadcastFullscreen;

        if (fullscreen && !wasFullscreen) {
            broadcastPreviousOrientation = getRequestedOrientation();
        }

        broadcastFullscreen = fullscreen;

        if (broadcastPlayer != null) {
            broadcastPlayer.setVolume(100);
        }

        FrameLayout.LayoutParams lp;

        if (fullscreen) {
            setRequestedOrientation(
                    android.content.pm.ActivityInfo.SCREEN_ORIENTATION_PORTRAIT
            );

            lp = new FrameLayout.LayoutParams(
                    FrameLayout.LayoutParams.MATCH_PARENT,
                    FrameLayout.LayoutParams.MATCH_PARENT,
                    Gravity.TOP | Gravity.LEFT
            );

            getWindow().getDecorView().setSystemUiVisibility(
                    View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY
                    | View.SYSTEM_UI_FLAG_FULLSCREEN
                    | View.SYSTEM_UI_FLAG_HIDE_NAVIGATION
                    | View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN
                    | View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION
                    | View.SYSTEM_UI_FLAG_LAYOUT_STABLE
            );

            if (broadcastHideButton != null) {
                broadcastHideButton.setVisibility(View.GONE);
            }

        } else {
            if (wasFullscreen) {
                int restore = broadcastPreviousOrientation >= 0
                        ? broadcastPreviousOrientation
                        : android.content.pm.ActivityInfo.SCREEN_ORIENTATION_UNSPECIFIED;

                setRequestedOrientation(restore);
                broadcastPreviousOrientation = -1;
            }

            lp = new FrameLayout.LayoutParams(
                    broadcastMiniWidth(), broadcastMiniHeight(),
                    Gravity.TOP | Gravity.LEFT
            );
            lp.setMargins(dp(10), dp(68), 0, 0);

            getWindow().getDecorView().setSystemUiVisibility(0);

            if (broadcastHideButton != null) {
                broadcastHideButton.setVisibility(View.VISIBLE);
            }
        }

        broadcastMiniBox.setLayoutParams(lp);
        broadcastMiniBox.bringToFront();
        updateMiniLogoLayout(broadcastLogo, fullscreen);
    }

    private void stopBroadcastMiniPlayer(String id) {
        if (id != null && !id.isEmpty() &&
                !id.equals(activeBroadcastId)) return;

        if (broadcastFullscreen) {
            getWindow().getDecorView().setSystemUiVisibility(0);
            int restore = broadcastPreviousOrientation >= 0
                    ? broadcastPreviousOrientation
                    : android.content.pm.ActivityInfo.SCREEN_ORIENTATION_UNSPECIFIED;
            setRequestedOrientation(restore);
        }

        broadcastPreviousOrientation = -1;
        broadcastFullscreen = false;

        if (broadcastPlayer != null) {
            try { broadcastPlayer.stop(); } catch (Exception ignored) {}
            try { broadcastPlayer.detachViews(); } catch (Exception ignored) {}
            try { broadcastPlayer.release(); } catch (Exception ignored) {}
        }

        if (broadcastLibVLC != null) {
            try { broadcastLibVLC.release(); } catch (Exception ignored) {}
        }

        if (broadcastProxy != null) {
            try { broadcastProxy.stop(); } catch (Exception ignored) {}
        }

        if (broadcastMiniBox != null && root != null)
            root.removeView(broadcastMiniBox);

        broadcastPlayer = null;
        broadcastPlayerView = null;
        broadcastLibVLC = null;
        broadcastProxy = null;
        broadcastMiniBox = null;
        broadcastLogo = null;
        broadcastHideButton = null;
        activeBroadcastId = "";

        if (tvLogo != null && shouldShowTvLogo())
            tvLogo.setVisibility(View.VISIBLE);
    }

                                        broadcastMonitorRunnable

            broadcastMonitorHandler.removeCallbacks(
                    broadcastMonitorRunnable
            );

            monitoredBroadcastId = id;

            broadcastMonitorHandler.postDelayed(
                    broadcastMonitorRunnable, 10000
            );
        }

                    id.equals(monitoredBroadcastId)) {

        public void setFavorite(String channelJson, boolean favorite) {
            try {
                JSONObject channel = new JSONObject(channelJson);
                String url = channel.optString("url", "");
                if (url.isEmpty()) return;

                android.content.SharedPreferences prefs =
                        getSharedPreferences("isa_tv_data", MODE_PRIVATE);

                JSONArray old = new JSONArray(prefs.getString("favorites", "[]"));
                JSONArray updated = new JSONArray();

                for (int i = 0; i < old.length(); i++) {
                    JSONObject item = old.optJSONObject(i);
                    if (item == null) continue;
                    if (!url.equals(item.optString("url", ""))) {
                        updated.put(item);
                    }
                }

                if (favorite) updated.put(channel);

                prefs.edit().putString("favorites", updated.toString()).apply();
            } catch (Exception e) {
                Log.e(TAG, "Favori kaydedilemedi", e);
            }
        }



        // İSAFLİX kalıcı üye oturumu: token yalnızca AndroidKeyStore
        // AES/GCM ile şifrelenmiş olarak SharedPreferences içinde tutulur.
        private static final String MEMBER_API =
                "https://member-panel.githuba85.workers.dev";
        private static final String MEMBER_PREFS =
                "isaflix_member_session";
        private static final String MEMBER_KEY_ALIAS =
                "isaflix_member_session_key";

        private android.content.SharedPreferences memberPrefs() {
            return getSharedPreferences(MEMBER_PREFS, MODE_PRIVATE);
        }

        private SecretKey memberSecretKey() throws Exception {
            KeyStore store = KeyStore.getInstance("AndroidKeyStore");
            store.load(null);

            java.security.Key existing = store.getKey(MEMBER_KEY_ALIAS, null);
            if (existing instanceof SecretKey) return (SecretKey) existing;

            KeyGenerator generator = KeyGenerator.getInstance(
                    KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore"
            );
            generator.init(new KeyGenParameterSpec.Builder(
                    MEMBER_KEY_ALIAS,
                    KeyProperties.PURPOSE_ENCRYPT |
                            KeyProperties.PURPOSE_DECRYPT
            ).setBlockModes(KeyProperties.BLOCK_MODE_GCM)
             .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
             .setRandomizedEncryptionRequired(true)
             .build());

            return generator.generateKey();
        }

        private void saveMemberToken(String token) throws Exception {
            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(Cipher.ENCRYPT_MODE, memberSecretKey());

            byte[] encrypted = cipher.doFinal(
                    token.getBytes(StandardCharsets.UTF_8)
            );

            memberPrefs().edit()
                    .putString("token_cipher",
                            Base64.encodeToString(encrypted, Base64.NO_WRAP))
                    .putString("token_iv",
                            Base64.encodeToString(cipher.getIV(), Base64.NO_WRAP))
                    .apply();
        }

        private String readMemberToken() throws Exception {
            String encryptedText = memberPrefs().getString("token_cipher", "");
            String ivText = memberPrefs().getString("token_iv", "");
            if (encryptedText.isEmpty() || ivText.isEmpty()) return null;

            byte[] encrypted = Base64.decode(encryptedText, Base64.NO_WRAP);
            byte[] iv = Base64.decode(ivText, Base64.NO_WRAP);

            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(
                    Cipher.DECRYPT_MODE,
                    memberSecretKey(),
                    new GCMParameterSpec(128, iv)
            );

            return new String(
                    cipher.doFinal(encrypted),
                    StandardCharsets.UTF_8
            );
        }

        private String memberDeviceId() {
            String id = memberPrefs().getString("device_id", "");
            if (id.isEmpty()) {
                id = java.util.UUID.randomUUID().toString();
                memberPrefs().edit().putString("device_id", id).apply();
            }
            return id;
        }

        private JSONObject memberHttp(
                String method, String path, String token, JSONObject body
        ) throws Exception {
            HttpURLConnection connection = null;
            try {
                URL url = new URL(MEMBER_API + path);
                connection = (HttpURLConnection) url.openConnection();
                connection.setRequestMethod(method);
                connection.setConnectTimeout(15000);
                connection.setReadTimeout(20000);
                connection.setRequestProperty("Accept", "application/json");
                connection.setRequestProperty("X-Device-ID", memberDeviceId());
                String memberName = memberPrefs().getString("member_username", "");
                if (memberName == null || memberName.isEmpty()) {
                    String query = url.getQuery();
                    if (query != null) {
                        for (String part : query.split("&")) {
                            String[] pair = part.split("=", 2);
                            if (pair.length == 2 && "username".equals(pair[0])) {
                                memberName = java.net.URLDecoder.decode(pair[1], "UTF-8");
                                break;
                            }
                        }
                    }
                }
                if (memberName != null && !memberName.isEmpty()) {
                    connection.setRequestProperty("X-Member-Username", memberName);
                }


                if (token != null && !token.isEmpty()) {
                    connection.setRequestProperty(
                            "Authorization", "Bearer " + token
                    );
                }

                if (body != null) {
                    connection.setDoOutput(true);
                    connection.setRequestProperty(
                            "Content-Type", "application/json; charset=utf-8"
                    );
                    byte[] bytes = body.toString().getBytes(StandardCharsets.UTF_8);
                    try (java.io.OutputStream output = connection.getOutputStream()) {
                        output.write(bytes);
                        output.flush();
                    }
                }

                int status = connection.getResponseCode();
                java.io.InputStream input = status >= 200 && status < 400
                        ? connection.getInputStream()
                        : connection.getErrorStream();

                String response = input == null ? "{}" : readStreamUtf8(input);
                JSONObject result;
                try {
                    result = new JSONObject(response);
                } catch (Exception ignored) {
                    result = new JSONObject();
                }
                result.put("_http_status", status);
                return result;
            } finally {
                if (connection != null) connection.disconnect();
            }
        }

        private String readStreamUtf8(java.io.InputStream input)
                throws Exception {
            StringBuilder result = new StringBuilder();

            try (
                java.io.BufferedReader reader =
                    new java.io.BufferedReader(
                        new java.io.InputStreamReader(
                            input,
                            java.nio.charset.StandardCharsets.UTF_8
                        ),
                        65536
                    )
            ) {
                char[] buffer = new char[32768];
                int count;

                while ((count = reader.read(buffer)) != -1) {
                    result.append(buffer, 0, count);
                }
            }

            return result.toString();
        }

        @JavascriptInterface
public void fetchText(String address, String callbackId) {
    networkExecutor.execute(() -> {
        String result = null;
        String error = null;

        for (int attempt = 1; attempt <= 3; attempt++) {
            HttpURLConnection connection = null;

            try {
                URL url = new URL(address);

                if (!"http".equalsIgnoreCase(url.getProtocol())
                        && !"https".equalsIgnoreCase(url.getProtocol())) {
                    throw new Exception("HTTP/HTTPS adresi gerekli.");
                }

                connection = (HttpURLConnection) url.openConnection();
                connection.setConnectTimeout(20000);
                connection.setReadTimeout(120000);
                connection.setRequestMethod("GET");
                connection.setInstanceFollowRedirects(true);
                connection.setRequestProperty("Accept", "*/*");
                connection.setRequestProperty("Accept-Encoding", "identity");
                connection.setRequestProperty(
                    "User-Agent",
                    "Mozilla/5.0 (Linux; Android 11) AppleWebKit/537.36 "
                    + "(KHTML, like Gecko) Chrome/120.0 Mobile Safari/537.36"
                );

                int code = connection.getResponseCode();

                if (code < 200 || code >= 300) {
                    throw new java.io.IOException("HTTP " + code);
                }

                StringBuilder output = new StringBuilder();

                try (BufferedReader reader = new BufferedReader(
                        new InputStreamReader(
                            connection.getInputStream(),
                            StandardCharsets.UTF_8))) {

                    String line;
                    while ((line = reader.readLine()) != null) {
                        output.append(line).append('\n');
}
                }

                result = output.toString();
                error = null;
                break;

            } catch (Exception e) {
                error = e.getMessage();
                Log.w(TAG, "Kaynak indirme denemesi " + attempt + "/3", e);

                if (attempt < 3) {
                    try {
                        Thread.sleep(attempt * 1200L);
                    } catch (InterruptedException interrupted) {
                        Thread.currentThread().interrupt();
                        error = "İşlem iptal edildi.";
                        break;
                    }
                }
            } finally {
                if (connection != null) {
                    connection.disconnect();
                }
            }
        }

        final String response = result;
        final String failure = error;
        final String safeId = callbackId == null
                ? "0"
                : callbackId.replaceAll("[^0-9]", "");

        runOnUiThread(() -> {
            if (webView == null) return;

            String js = "window.__isaFetchDone("
                    + safeId + ","
                    + JSONObject.quote(response) + ","
                    + JSONObject.quote(failure) + ");";

            webView.evaluateJavascript(js, null);
        });
    });
}

    @Override
    public void onBackPressed() {
        if (broadcastFullscreen) {
            setBroadcastFullscreen(false);
            return;
        }
        if (webView == null) {
            handleExitBack();
            return;
        }

        webView.evaluateJavascript(
                "(function(){return window.isaHandleBack ? window.isaHandleBack() : false;})()",
                value -> {
                    if ("true".equals(value)) {
                        backPressedAt = 0L;
                    } else {
                        handleExitBack();
                    }
                }
        );
    }

    private void handleExitBack() {
        long now = System.currentTimeMillis();
        if (now - backPressedAt < 2000L) {
            finish();
            return;
        }

        backPressedAt = now;
        Toast.makeText(
                this,
                "Çıkmak için geri tuşuna tekrar basın",
                Toast.LENGTH_SHORT
        ).show();
    }

    @Override
    protected void onDestroy() {
        if (fileCallback != null) {
            fileCallback.onReceiveValue(null);
            fileCallback = null;
        }

        stopBroadcastMiniPlayer(null);
        networkExecutor.shutdownNow();

        if (webView != null) {
            webView.destroy();
            webView = null;
        }

    @Override
    public boolean dispatchKeyEvent(android.view.KeyEvent event) {
        int key = event.getKeyCode();
        boolean down = event.getAction() == android.view.KeyEvent.ACTION_DOWN;
        boolean portrait = getResources().getConfiguration().orientation
                == Configuration.ORIENTATION_PORTRAIT;

        if (!portrait && down && event.getRepeatCount() == 0) {
            if (channelDialog != null && channelDialog.isShowing()) {
                if (key == android.view.KeyEvent.KEYCODE_DPAD_LEFT) {
                    View v = channelDialog.findViewById(ISA_GROUP_PREVIOUS_ID);
                    if (v != null) v.performClick();
                    return true;
                }
                if (key == android.view.KeyEvent.KEYCODE_DPAD_RIGHT) {
                    View v = channelDialog.findViewById(ISA_GROUP_NEXT_ID);
                    if (v != null) v.performClick();
                    return true;
                }
                if (key == android.view.KeyEvent.KEYCODE_BACK) {
                    channelDialog.dismiss();
                    return true;
                }
            } else if (key == android.view.KeyEvent.KEYCODE_DPAD_CENTER
                    || key == android.view.KeyEvent.KEYCODE_ENTER
                    || key == android.view.KeyEvent.KEYCODE_NUMPAD_ENTER) {
                showChannelList();
                return true;
            }
        }

        boolean next = key == android.view.KeyEvent.KEYCODE_DPAD_UP
                || key == android.view.KeyEvent.KEYCODE_CHANNEL_UP;
        boolean previous = key == android.view.KeyEvent.KEYCODE_DPAD_DOWN
                || key == android.view.KeyEvent.KEYCODE_CHANNEL_DOWN;

        if (next || previous) {
            if (down && event.getRepeatCount() == 0) {
                if (channelDialog != null && channelDialog.isShowing()) {
                    return super.dispatchKeyEvent(event);
                }
                finishChannelSwipe(next ? 1 : -1, false);
            }
            return true;
        }
        return super.dispatchKeyEvent(event);
    }

                boolean portrait =

                switch (event.getAction()) {
                    case android.view.MotionEvent.ACTION_DOWN:

                        if (channelTransitioning) return true;

                        float dy = event.getY() - startY;

    private void initializePlayer() {
        try {
            ArrayList<String> options = new ArrayList<>();
            options.add("--no-drop-late-frames");
            options.add("--no-skip-frames");
            options.add("--avcodec-hw=any");
            options.add("--network-caching=1500");
            options.add("--http-reconnect");
            libVLC = new LibVLC(this, options);
            vlcPlayer = new MediaPlayer(libVLC);
            vlcPlayer.attachViews(playerView, null, false, true);
            vlcPlayer.setEventListener(event -> handler.post(() -> {
                if (isFinishing()) return;
                if (event.type == MediaPlayer.Event.Playing) {
                    if (loadingSpinner != null)
                        loadingSpinner.setVisibility(View.GONE);
                    if (status != null) status.setVisibility(View.GONE);
                } else if (event.type == MediaPlayer.Event.Opening) {
                    if (loadingSpinner != null)
                        loadingSpinner.setVisibility(View.VISIBLE);
                } else if (event.type == MediaPlayer.Event.Buffering) {
                    // Buffering olayı oynatma başladıktan sonra
                    // yükleme göstergesini yeniden açmaz.
                } else if (event.type == MediaPlayer.Event.EncounteredError) {
                    if (loadingSpinner != null)
                        loadingSpinner.setVisibility(View.GONE);
                    if (status != null) {
                        status.setText("Yayın açılamadı.");
                        status.setVisibility(View.VISIBLE);
                    }
                } else if (event.type == MediaPlayer.Event.EndReached &&
                           selectedIndex >= 0 && selectedIndex < channels.size() &&
                           channels.get(selectedIndex).isLive) {
                    handler.postDelayed(this::playSelected, 1500);
                }
            }));
            streamProxy = new StreamProxyServer();
        } catch (Exception e) {
            Log.e(TAG, "VLC başlatılamadı", e);
            if (status != null) {
                status.setText("Oynatıcı başlatılamadı.");
                status.setVisibility(View.VISIBLE);
            }
        }
    }


    

    private void prepareChannel(int index) {
        if (index < 0 || index >= channels.size() ||
                vlcPlayer == null || libVLC == null) return;
        Channel channel = channels.get(index);
        try {
            String playbackUrl = streamProxy.register(channel.url, channel.headers);
            vlcPlayer.stop();
            Media media = new Media(libVLC, android.net.Uri.parse(playbackUrl));
            media.addOption(":network-caching=1500");
            media.addOption(":clock-jitter=0");
            media.addOption(":http-reconnect");
            media.addOption(":avcodec-hw=any");
            vlcPlayer.setVolume(100);
            vlcPlayer.setMedia(media);
            media.release();
            vlcPlayer.play();
        } catch (Exception e) {
            Log.e(TAG, "VLC yayını hazırlanamadı", e);
            if (loadingSpinner != null)
                loadingSpinner.setVisibility(View.GONE);
            if (status != null) {
                status.setText("Yayın hazırlanamadı.");
                status.setVisibility(View.VISIBLE);
            }
        }
    }

    private void animateViewsBack(boolean horizontal) {

                    .translationX(horizontal ? 0f : portraitPanel.getTranslationX())

    private void finishChannelSwipe(int direction, boolean horizontal) {
        if (channels.isEmpty() || channelTransitioning ||
                playerView == null) return;

        channelTransitioning = true;

        final int distance = horizontal ? Math.max(1, playerView.getWidth()) : Math.max(
                playerView.getHeight(), getResources().getDisplayMetrics().heightPixels);
        final float exit = direction > 0 ? -distance : distance;
        final float enter = -exit;

        playerView.animate().cancel();

        if (portraitPanel != null) {
            portraitPanel.animate().cancel();
        }
        if (!horizontal && logo != null) logo.animate().cancel();

        playerView.animate()
                .translationX(horizontal ? exit : 0f)
                .translationY(horizontal ? 0f : exit)
                .setDuration(150)
                .withEndAction(() -> {
                    selectedIndex += direction;

                    if (selectedIndex >= channels.size()) selectedIndex = 0;
                    if (selectedIndex < 0) selectedIndex = channels.size() - 1;

                    playSelected();

                    playerView.setTranslationX(horizontal ? enter : 0f);
                    playerView.setTranslationY(horizontal ? 0f : enter);

                    if (portraitPanel != null) {
                        portraitPanel.setTranslationX(horizontal ? enter : 0f);
                        portraitPanel.setTranslationY(horizontal ? 0f : enter);
                    }
                    if (logo != null && !horizontal) {
                        logo.setTranslationY(enter);
                        logo.animate().translationY(0f).setDuration(220).start();
                    }

                    playerView.animate()
                            .translationX(0f)
                            .translationY(0f)
                            .setDuration(220)
                            .withEndAction(() -> {
                                channelTransitioning = false;
                            })
                            .start();

                    if (portraitPanel != null) {
                        portraitPanel.animate()
                                .translationX(0f)
                                .translationY(0f)
                                .setDuration(220)
                                .start();
                    }
                })
                .start();

        if (portraitPanel != null) {
            portraitPanel.animate()
                    .translationX(horizontal ? exit : 0f)
                    .translationY(horizontal ? 0f : exit)
                    .setDuration(150)
                    .start();
        }
        if (logo != null && !horizontal) {
            logo.animate().translationY(exit).setDuration(150).start();
        }
    }

    private void playSelected() {
        if (vlcPlayer == null || selectedIndex < 0 ||
                selectedIndex >= channels.size()) return;
        Channel channel = channels.get(selectedIndex);
        updatePortraitOverlay();
        showLandscapeInfo();
        if (channel.url == null ||
                !channel.url.matches("(?i)^https?://.*")) {
            if (!channel.itemId.isEmpty() || !channel.channelId.isEmpty()) {
                resolveMemberPlayback(channel);
                return;
            }
            if (loadingSpinner != null)
                loadingSpinner.setVisibility(View.GONE);
            if (status != null) {
                status.setText("İçerik açılamıyor.");
                status.setVisibility(View.VISIBLE);

            return;

        if (loadingSpinner != null)
            loadingSpinner.setVisibility(View.VISIBLE);
        if (status != null) status.setVisibility(View.GONE);
        setTitle(channel.name);
        prepareChannel(selectedIndex);
    }

private boolean isTelevisionDevice() {
    int type = getResources().getConfiguration().uiMode
            & Configuration.UI_MODE_TYPE_MASK;

    return type == Configuration.UI_MODE_TYPE_TELEVISION
            || getPackageManager().hasSystemFeature(
                    android.content.pm.PackageManager.FEATURE_LEANBACK
            );
}

private boolean isPhoneDevice() {
    if (isTelevisionDevice()) return false;

    int smallestWidth =
            getResources().getConfiguration().smallestScreenWidthDp;

    if (smallestWidth > 0) {
        return smallestWidth < 600;
    }

    float density = getResources().getDisplayMetrics().density;
    float widthDp =
            getResources().getDisplayMetrics().widthPixels / density;

    return widthDp < 600;
}

private boolean isCurrentChannelFavorite() {
    if (selectedIndex < 0 || selectedIndex >= channels.size()) return false;

    String url = channels.get(selectedIndex).url;
    try {
        JSONArray favorites = new JSONArray(
                getSharedPreferences("isa_tv_data", MODE_PRIVATE)
                        .getString("favorites", "[]"));

        for (int i = 0; i < favorites.length(); i++) {
            JSONObject item = favorites.optJSONObject(i);
            if (item != null && url.equals(item.optString("url", ""))) {
                return true;
            }
        }
    } catch (Exception e) {
        Log.e(TAG, "Favoriler okunamadı", e);
    }
    return false;
}

private void updatePortraitOverlay() {
    if (portraitChannelName == null || favoriteButton == null) return;

    if (portraitPanel != null) portraitPanel.setVisibility(portrait ? View.VISIBLE : View.GONE);

    portraitChannelName.setVisibility(portrait ? View.VISIBLE : View.GONE);

private void toggleCurrentFavorite() {
    if (selectedIndex < 0 || selectedIndex >= channels.size()) return;

    Channel channel = channels.get(selectedIndex);
    boolean add = !isCurrentChannelFavorite();

    try {
        JSONObject item = new JSONObject();
        item.put("name", channel.name == null ? "İsimsiz kanal" : channel.name);
        item.put("url", channel.url);
        item.put("group", channel.group == null ? "Genel" : channel.group);
        item.put("logo", channel.logo == null ? "" : channel.logo);
        item.put("is_live", channel.isLive);
        item.put("season", channel.season);
        item.put("episode", channel.episode);

        android.content.SharedPreferences prefs =
                getSharedPreferences("isa_tv_data", MODE_PRIVATE);

        JSONArray old = new JSONArray(prefs.getString("favorites", "[]"));
        JSONArray updated = new JSONArray();

        for (int i = 0; i < old.length(); i++) {
            JSONObject existing = old.optJSONObject(i);
            if (existing == null) continue;

            if (!channel.url.equals(existing.optString("url", ""))) {
                updated.put(existing);
            }
        }

        if (add) updated.put(item);

        prefs.edit().putString("favorites", updated.toString()).apply();
        updatePortraitOverlay();

    } catch (Exception e) {
        Log.e(TAG, "Favori değiştirilemedi", e);
    }
}

        }

        updatePortraitOverlay();
        if (landscapeInfo != null && isPortraitMode()) {
            handler.removeCallbacks(hideLandscapeInfoCard);
            landscapeInfo.setVisibility(View.GONE);
        }


        applyPreloadPolicy();

        if (channelButton != null) {
            channelButton.setVisibility(View.GONE);
        }

    }

    @Override
    protected void onPause() {
        super.onPause();
        resumeAfterPause = vlcPlayer != null && vlcPlayer.isPlaying();
        if (resumeAfterPause) vlcPlayer.pause();
    }

    @Override
    protected void onResume() {
        super.onResume();
        if (resumeAfterPause && vlcPlayer != null) {
            resumeAfterPause = false;
            vlcPlayer.play();
        }
    }

    @Override
    protected void onDestroy() {
        if (activeBroadcastPlayer == this) activeBroadcastPlayer = null;
        handler.removeCallbacksAndMessages(null);
        if (vlcPlayer != null) {
            try { vlcPlayer.stop(); } catch (Exception ignored) {}
            try { vlcPlayer.detachViews(); } catch (Exception ignored) {}
            try { vlcPlayer.release(); } catch (Exception ignored) {}
            vlcPlayer = null;
        }
        if (libVLC != null) {
            try { libVLC.release(); } catch (Exception ignored) {}
            libVLC = null;
        }
        if (streamProxy != null) {
            try { streamProxy.stop(); } catch (Exception ignored) {}
            streamProxy = null;
        }
        super.onDestroy();
    }

        boolean portrait = getResources().getConfiguration().orientation
                == Configuration.ORIENTATION_PORTRAIT;

            if (channelDialog != null && channelDialog.isShowing()) {
                if (key == android.view.KeyEvent.KEYCODE_DPAD_LEFT) {
                    View v = channelDialog.findViewById(ISA_GROUP_PREVIOUS_ID);
                    if (v != null) v.performClick();
                    return true;
                }
                if (key == android.view.KeyEvent.KEYCODE_DPAD_RIGHT) {
                    View v = channelDialog.findViewById(ISA_GROUP_NEXT_ID);
                    if (v != null) v.performClick();
                    return true;
                }
                if (key == android.view.KeyEvent.KEYCODE_BACK) {
                    channelDialog.dismiss();
                    return true;
                }
            } else if (key == android.view.KeyEvent.KEYCODE_DPAD_CENTER
                    || key == android.view.KeyEvent.KEYCODE_ENTER
                    || key == android.view.KeyEvent.KEYCODE_NUMPAD_ENTER) {
                showChannelList();
                return true;
            }
        }

        boolean next = key == android.view.KeyEvent.KEYCODE_DPAD_UP
                || key == android.view.KeyEvent.KEYCODE_CHANNEL_UP;
        boolean previous = key == android.view.KeyEvent.KEYCODE_DPAD_DOWN
                || key == android.view.KeyEvent.KEYCODE_CHANNEL_DOWN;

        if (next || previous) {
            if (down && event.getRepeatCount() == 0) {
                if (channelDialog != null && channelDialog.isShowing()) {
                    return super.dispatchKeyEvent(event);
                }
                finishChannelSwipe(next ? 1 : -1);
            }
            return true;
        }
        return super.dispatchKeyEvent(event);
    }

                boolean portrait =

    private void initializePlayer() {
        try {
            ArrayList<String> options = new ArrayList<>();
            options.add("--no-drop-late-frames");
            options.add("--no-skip-frames");
            options.add("--avcodec-hw=any");
            options.add("--network-caching=1500");
            options.add("--http-reconnect");
            libVLC = new LibVLC(this, options);
            vlcPlayer = new MediaPlayer(libVLC);
            vlcPlayer.attachViews(playerView, null, false, true);
            vlcPlayer.setEventListener(event -> handler.post(() -> {
                if (isFinishing()) return;
                if (event.type == MediaPlayer.Event.Playing) {
                    if (loadingSpinner != null)
                        loadingSpinner.setVisibility(View.GONE);
                    if (status != null) status.setVisibility(View.GONE);
                } else if (event.type == MediaPlayer.Event.Opening) {
                    if (loadingSpinner != null)
                        loadingSpinner.setVisibility(View.VISIBLE);
                } else if (event.type == MediaPlayer.Event.Buffering) {
                    // Buffering olayı oynatma başladıktan sonra
                    // yükleme göstergesini yeniden açmaz.
                } else if (event.type == MediaPlayer.Event.EncounteredError) {
                    if (loadingSpinner != null)
                        loadingSpinner.setVisibility(View.GONE);
                    if (status != null) {
                        status.setText("Yayın açılamadı.");
                        status.setVisibility(View.VISIBLE);
                    }
                } else if (event.type == MediaPlayer.Event.EndReached &&
                           selectedIndex >= 0 && selectedIndex < channels.size() &&
                           channels.get(selectedIndex).isLive) {
                    handler.postDelayed(this::playSelected, 1500);
                }
            }));
            streamProxy = new StreamProxyServer();
        } catch (Exception e) {
            Log.e(TAG, "VLC başlatılamadı", e);
            if (status != null) {
                status.setText("Oynatıcı başlatılamadı.");
                status.setVisibility(View.VISIBLE);
            }
        }
    }


    

    private void prepareChannel(int index) {
        if (index < 0 || index >= channels.size() ||
                vlcPlayer == null || libVLC == null) return;
        Channel channel = channels.get(index);
        try {
            String playbackUrl = streamProxy.register(channel.url, channel.headers);
            vlcPlayer.stop();
            Media media = new Media(libVLC, android.net.Uri.parse(playbackUrl));
            media.addOption(":network-caching=1500");
            media.addOption(":clock-jitter=0");
            media.addOption(":http-reconnect");
            media.addOption(":avcodec-hw=any");
            vlcPlayer.setVolume(100);
            vlcPlayer.setMedia(media);
            media.release();
            vlcPlayer.play();
        } catch (Exception e) {
            Log.e(TAG, "VLC yayını hazırlanamadı", e);
            if (loadingSpinner != null)
                loadingSpinner.setVisibility(View.GONE);
            if (status != null) {
                status.setText("Yayın hazırlanamadı.");
                status.setVisibility(View.VISIBLE);
            }
        }
    }

    private void finishChannelSwipe(int direction, boolean horizontal) {
        if (channels.isEmpty() || channelTransitioning ||
                playerView == null) return;

        channelTransitioning = true;

        final int distance = horizontal ? Math.max(1, playerView.getWidth()) : Math.max(
                playerView.getHeight(), getResources().getDisplayMetrics().heightPixels);
        final float exit = direction > 0 ? -distance : distance;
        final float enter = -exit;

        playerView.animate().cancel();

        if (portraitPanel != null) {
            portraitPanel.animate().cancel();
        }

        playerView.animate()
                .translationX(horizontal ? exit : 0f)
                .translationY(horizontal ? 0f : exit)
                .setDuration(150)
                .withEndAction(() -> {
                    selectedIndex += direction;

                    if (selectedIndex >= channels.size()) selectedIndex = 0;
                    if (selectedIndex < 0) selectedIndex = channels.size() - 1;

                    playSelected();

                    playerView.setTranslationX(horizontal ? enter : 0f);
                    playerView.setTranslationY(horizontal ? 0f : enter);

                    if (portraitPanel != null) {
                        portraitPanel.setTranslationX(horizontal ? enter : 0f);
                        portraitPanel.setTranslationY(horizontal ? 0f : enter);
                    }

                    playerView.animate()
                            .translationX(0f)
                            .translationY(0f)
                            .setDuration(220)
                            .withEndAction(() -> {
                                channelTransitioning = false;
                            })
                            .start();

                    if (portraitPanel != null) {
                        portraitPanel.animate()
                                .translationX(0f)
                                .translationY(0f)
                                .setDuration(220)
                                .start();
                    }
                })
                .start();

        if (portraitPanel != null) {
            portraitPanel.animate()
                    .translationX(horizontal ? exit : 0f)
                    .translationY(horizontal ? 0f : exit)
                    .setDuration(150)
                    .start();
        }
    }

    

    private static String magEncode(String value) throws Exception {
        return URLEncoder.encode(value == null ? "" : value, "UTF-8");
    }

    private static String cleanMagCommand(String value) {
        if (value == null) return "";
        return value.trim()
                .replaceFirst("(?i)^(ffmpeg|auto)\\s+", "")
                .trim();
    }

    private String resolveMagUrl(Channel channel) throws Exception {
        String direct = cleanMagCommand(channel.magCmd);

        if (direct.matches("(?i)^https?://.*")) {
            return direct;
        }

        String portal = channel.magHost.replaceAll("/+$", "")
                + "/portal.php";

        String query = "type=itv&action=create_link"
                + "&cmd=" + magEncode(channel.magCmd)
                + "&series=0&forced_storage=&disable_ad=0"
                + "&JsHttpRequest=1-xml"
                + "&mac=" + magEncode(channel.magMac);

        HttpURLConnection connection = null;

        try {
            connection = (HttpURLConnection)
                    new URL(portal + "?" + query).openConnection();

            connection.setRequestMethod("GET");
            connection.setConnectTimeout(20000);
            connection.setReadTimeout(30000);
            connection.setInstanceFollowRedirects(true);

            for (Map.Entry<String, String> entry
                    : channel.headers.entrySet()) {
                connection.setRequestProperty(
                        entry.getKey(), entry.getValue());
            }

            connection.setRequestProperty(
                    "Authorization", "Bearer " + channel.magToken);

            connection.setRequestProperty(
                    "Cookie",
                    "mac=" + channel.magMac
                    + "; stb_lang=en; timezone=Europe/Istanbul");

            connection.setRequestProperty("Accept", "*/*");

            int code = connection.getResponseCode();

            InputStream stream = code >= 200 && code < 400
                    ? connection.getInputStream()
                    : connection.getErrorStream();

            if (stream == null) {
                throw new Exception("MAG portalı yanıt vermedi.");
            }

            StringBuilder body = new StringBuilder();

            try (BufferedReader reader = new BufferedReader(
                    new InputStreamReader(
                            stream, StandardCharsets.UTF_8))) {
                String line;
                while ((line = reader.readLine()) != null) {
                    body.append(line);
                }
            }

            if (code < 200 || code >= 400) {
                throw new Exception(
                        "MAG bağlantı isteği başarısız: " + code);
            }

            JSONObject root = new JSONObject(body.toString());
            JSONObject data = root.optJSONObject("js");

            if (data == null) data = root;

            String value = data.optString(
                    "cmd", data.optString("url", ""));

            JSONObject nested = data.optJSONObject("data");

            if ((value == null || value.isEmpty())
                    && nested != null) {
                value = nested.optString(
                        "cmd", nested.optString("url", ""));
            }

            value = cleanMagCommand(value);

            if (!value.matches("(?i)^https?://.*")) {
                throw new Exception(
                        "MAG portalı oynatılabilir bağlantı döndürmedi.");
            }

            return value;

        } finally {
            if (connection != null) connection.disconnect();
        }
    }

    private void resolveMagForPlayback(Channel channel, int index) {
        final int requestId = ++magPlaybackRequest;

        if (loadingSpinner != null) {
            loadingSpinner.setVisibility(View.VISIBLE);
        }

        if (status != null) {
            status.setText("Yayın hazırlanıyor...");
            status.setVisibility(View.VISIBLE);
        }

        magExecutor.execute(() -> {
            String resolved = null;
            String errorMessage = null;

            try {
                resolved = resolveMagUrl(channel);
            } catch (Exception e) {
                errorMessage = e.getMessage() == null
                        ? "MAG yayını açılamadı."
                        : e.getMessage();
            }

            final String finalUrl = resolved;
            final String finalError = errorMessage;

            handler.post(() -> {
                if (isFinishing()
                        || requestId != magPlaybackRequest
                        || selectedIndex != index
                        || index < 0
                        || index >= channels.size()
                        || channels.get(index) != channel) {
                    return;
                }

                if (finalUrl == null) {
                    if (loadingSpinner != null) {
                        loadingSpinner.setVisibility(View.GONE);
                    }

                    if (status != null) {
                        status.setText(finalError);
                        status.setVisibility(View.VISIBLE);
                    }
                    return;
                }

                channel.url = finalUrl;

                if (status != null) {
                    status.setVisibility(View.GONE);
                }

                playSelected();
            });
        });
    }

    private void playSelected() {
        if (vlcPlayer == null || selectedIndex < 0 ||
                selectedIndex >= channels.size()) return;
        Channel channel = channels.get(selectedIndex);
        showLandscapeInfo();
        if (channel.mag && (channel.url == null ||
                !channel.url.matches("(?i)^https?://.*"))) {
            resolveMagForPlayback(channel, selectedIndex);
            return;
        }
        updatePortraitOverlay();
        if (loadingSpinner != null)
            loadingSpinner.setVisibility(View.VISIBLE);
        if (status != null) status.setVisibility(View.GONE);
        setTitle(channel.name);
        prepareChannel(selectedIndex);
    }

private boolean isPortraitMode() {
    return getResources().getConfiguration().orientation
            == Configuration.ORIENTATION_PORTRAIT;

private boolean isTelevisionDevice() {
    int type = getResources().getConfiguration().uiMode
            & Configuration.UI_MODE_TYPE_MASK;

    return type == Configuration.UI_MODE_TYPE_TELEVISION
            || getPackageManager().hasSystemFeature(
                    android.content.pm.PackageManager.FEATURE_LEANBACK
            );
}

private boolean isPhoneDevice() {
    if (isTelevisionDevice()) return false;

    int smallestWidth =
            getResources().getConfiguration().smallestScreenWidthDp;

    if (smallestWidth > 0) {
        return smallestWidth < 600;
    }

    float density = getResources().getDisplayMetrics().density;
    float widthDp =
            getResources().getDisplayMetrics().widthPixels / density;

    return widthDp < 600;
}

private boolean isCurrentChannelFavorite() {
    if (selectedIndex < 0 || selectedIndex >= channels.size()) return false;

    String url = channels.get(selectedIndex).url;
    try {
        JSONArray favorites = new JSONArray(
                getSharedPreferences("isa_tv_data", MODE_PRIVATE)
                        .getString("favorites", "[]"));

        for (int i = 0; i < favorites.length(); i++) {
            JSONObject item = favorites.optJSONObject(i);
            if (item != null && url.equals(item.optString("url", ""))) {
                return true;
            }
        }
    } catch (Exception e) {
        Log.e(TAG, "Favoriler okunamadı", e);
    }
    return false;
}

private void updatePortraitOverlay() {
    if (portraitChannelName == null || favoriteButton == null) return;

    boolean portrait = isPortraitMode();
    if (portraitPanel != null) portraitPanel.setVisibility(portrait ? View.VISIBLE : View.GONE);

    portraitChannelName.setVisibility(portrait ? View.VISIBLE : View.GONE);
    boolean liveItem = selectedIndex >= 0 && selectedIndex < channels.size()
            && channels.get(selectedIndex).isLive;
    portraitLiveBadge.setVisibility(portrait && liveItem ? View.VISIBLE : View.GONE);
    if (portraitEpisodeInfo != null) portraitEpisodeInfo.setVisibility(portrait ? View.VISIBLE : View.GONE);
    favoriteButton.setVisibility(portrait ? View.VISIBLE : View.GONE);

    if (!portrait) return;

    if (selectedIndex >= 0 && selectedIndex < channels.size()) {
        Channel channel = channels.get(selectedIndex);
        portraitChannelName.setText(
                channel.name == null ? "İSA TV HD" : channel.name);
        if (portraitEpisodeInfo != null) {

                        (channel.episode > 0 ? "  ·  " + channel.episode + ". Bölüm" : "");

                    : "Favorilere ekle");
}

private void toggleCurrentFavorite() {
    if (selectedIndex < 0 || selectedIndex >= channels.size()) return;

    Channel channel = channels.get(selectedIndex);
    boolean add = !isCurrentChannelFavorite();

    try {
        JSONObject item = new JSONObject();
        item.put("name", channel.name == null ? "İsimsiz kanal" : channel.name);
        item.put("url", channel.url);
        item.put("group", channel.group == null ? "Genel" : channel.group);
        item.put("logo", channel.logo == null ? "" : channel.logo);
        item.put("is_live", channel.isLive);
        item.put("season", channel.season);
        item.put("episode", channel.episode);

        android.content.SharedPreferences prefs =
                getSharedPreferences("isa_tv_data", MODE_PRIVATE);

        JSONArray old = new JSONArray(prefs.getString("favorites", "[]"));
        JSONArray updated = new JSONArray();

        for (int i = 0; i < old.length(); i++) {
            JSONObject existing = old.optJSONObject(i);
            if (existing == null) continue;

            if (!channel.url.equals(existing.optString("url", ""))) {
                updated.put(existing);
            }
        }

        if (add) updated.put(item);

        prefs.edit().putString("favorites", updated.toString()).apply();
        updatePortraitOverlay();

    } catch (Exception e) {
        Log.e(TAG, "Favori değiştirilemedi", e);
    }
}

    private void checkForUpdate() {
        if (isFinishing() || launched || checkStarted) return;
        checkStarted = true;

        executor.execute(() -> {
            int remoteCode = 0;
            String remoteName = "";
            String remoteUrl = "";
            String notes = "";

            HttpURLConnection connection = null;
            try {
                URL url = new URL(
                    "https://isatvhd.online/app-update"
                );
                connection = (HttpURLConnection) url.openConnection();
                connection.setRequestMethod("GET");
                connection.setConnectTimeout(10000);
                connection.setReadTimeout(12000);
                connection.setRequestProperty("Accept", "application/json");
                connection.setUseCaches(false);

                int status = connection.getResponseCode();
                if (status >= 200 && status < 300) {
                    try (InputStream input = connection.getInputStream();
                         ByteArrayOutputStream output = new ByteArrayOutputStream()) {
                        byte[] buffer = new byte[4096];
                        int count;
                        while ((count = input.read(buffer)) != -1) {
                            output.write(buffer, 0, count);
                            if (output.size() > 1024 * 1024) break;
                        }

                        JSONObject data = new JSONObject(
                            output.toString(StandardCharsets.UTF_8.name())
                        );
                        if (data.optBoolean("ok", false)) {
                            remoteCode = data.optInt("version_code", 0);
                            remoteName = data.optString("version_name", "");
                            remoteUrl = data.optString("apk_url", "");
                            notes = data.optString("notes", "");
                        }
                    }
                }
            } catch (Exception ignored) {
                // Sunucuya ulaşılamazsa mevcut uygulama açılmaya devam eder.
            } finally {
                if (connection != null) connection.disconnect();
            }

            final int code = remoteCode;
            final String name = remoteName;
            final String apk = remoteUrl;
            final String description = notes;

            runOnUiThread(() -> {
                checkStarted = false;
                if (isFinishing() || launched) return;

                if (code > BuildConfig.VERSION_CODE &&
                    apk.startsWith("https://")) {
                    requiredVersionCode = code;
                    updateUrl = apk;
                    updateNotes = description;
                    updateRequired = true;
                    showRequiredUpdate(name);
                } else {
                    openMain();
                }
            });
        });
    }

    private void showRequiredUpdate(String versionName) {
        if (isFinishing()) return;

        String message = "Yeni sürüm: " + versionName +
                "\nYüklü sürüm: " + BuildConfig.VERSION_NAME +
                "\n\n" + updateNotes +
                "\n\nUygulamayı kullanmaya devam etmek için güncelleme yapmalısınız.";

        if (updateDialog != null && updateDialog.isShowing()) return;

        updateDialog = new AlertDialog.Builder(this)
            .setTitle("Güncelleme gerekli")
            .setMessage(message)
            .setCancelable(false)
            .setPositiveButton("Güncelle", (dialog, which) -> openUpdatePage())
            .setNegativeButton("Tekrar kontrol et", null)
            .create();

        updateDialog.setCanceledOnTouchOutside(false);
        updateDialog.setOnShowListener(dialog -> {
            updateDialog.getButton(DialogInterface.BUTTON_NEGATIVE)
                .setOnClickListener(v -> {
                    updateDialog.dismiss();
                    checkStarted = false;
                    checkForUpdate();
                });
        });
        updateDialog.show();
        updateDialog.setCanceledOnTouchOutside(false);
    }

    private void openUpdatePage() {
        try {
            Intent intent = new Intent(Intent.ACTION_VIEW, Uri.parse(updateUrl));
            startActivity(intent);
        } catch (Exception ignored) {
            if (!isFinishing()) {
                new AlertDialog.Builder(this)
                    .setMessage("Güncelleme adresi açılamadı. Lütfen tekrar deneyin.")
                    .setPositiveButton("Tamam", null)
                    .show();
            }
        }
    }

    @Override
    protected void onResume() {
        super.onResume();

        if (updateRequired && Build.VERSION.SDK_INT >= 1 &&
            BuildConfig.VERSION_CODE >= requiredVersionCode) {
            updateRequired = false;
            if (updateDialog != null) updateDialog.dismiss();
            openMain();
        }
    }

    private void openMain() {
        if (launched || isFinishing()) return;
        launched = true;
        startActivity(new Intent(SplashActivity.this, MainActivity.class));
        overridePendingTransition(0, 0);
        finish();
    }

    @Override
    protected void onDestroy() {
        handler.removeCallbacksAndMessages(null);
        executor.shutdownNow();
        super.onDestroy();
    }

            );
            const logo = String(
                item.logo || item.logo_url || item.image_url || item.poster_url || ''
            );

                id: String(item.id || ''),

                poster_url: logo,

            };

            movies: Array.isArray(result.movies) ? result.movies : [],
            series: Array.isArray(result.series) ? result.series : [],
            live_channels: channels,
            live_categories: Array.isArray(result.live_categories)
                ? result.live_categories
                : [...new Set(channels.map(x => x.group))].map(name => ({name})),

function nativeCacheGet(key) {
    if (!window.ISATV || !window.ISATV.cacheGet) {
        try {
            return Promise.resolve(localStorage.getItem('isa-cache:'+key));
        } catch(e) {
            return Promise.resolve(null);
        }
    }

    return new Promise((resolve,reject) => {
        const id = ++cacheCounter;
        window.__isaCacheCallbacks[id] = {resolve,reject};
        window.ISATV.cacheGet(key,String(id));
    });
}

function nativeCachePut(key,value) {
    if (window.ISATV && window.ISATV.cachePut) {
        window.ISATV.cachePut(key,value);
    } else {
        try {
            localStorage.setItem('isa-cache:'+key,value);
        } catch(e) {}
    }
}

function nativeFetch(url) {
    if (!window.ISATV || !window.ISATV.fetchText) {
        return fetch(url).then(response => {
            if (!response.ok) throw new Error('HTTP ' + response.status);
            return response.text();
        });
    }
    return new Promise((resolve,reject) => {
        const id = ++fetchCounter;
        window.__isaFetchCallbacks[id] = {resolve,reject};
        window.ISATV.fetchText(url,String(id));
    });
}

function nativeFetchWithHeaders(url,headers={}) {
    if (!window.ISATV || !window.ISATV.fetchTextWithHeaders) {
        return fetch(url,{headers}).then(response => {
            if (!response.ok) throw new Error('HTTP ' + response.status);
            return response.text();
        });
    }
    return new Promise((resolve,reject) => {
        const id = ++fetchCounter;
        window.__isaFetchCallbacks[id] = {resolve,reject};
        window.ISATV.fetchTextWithHeaders(url,JSON.stringify(headers || {}),String(id));
    });
}

function getTextWithHeaders(url,headers={}) {
    return nativeFetchWithHeaders(url,headers);
}

async function getText(url,forceRefresh=false) {
    let cached = null;

    try {
        cached = await nativeCacheGet(url);
    } catch(e) {}

    if (!forceRefresh && cached) return cached;

    try {
        const fresh = await nativeFetch(url);
        nativeCachePut(url,fresh);
        return fresh;
    } catch(e) {
        if (cached) return cached;
        throw e;
    }
}

function isHttp(url) {
    return /^https?:\/\//i.test(String(url||''));
}


function isDirectMedia(url) {
    if (!isHttp(url)) return false;
    try {
        const path = new URL(url).pathname.toLowerCase();
        return /\.(m3u8|mp4|mpd|mp3|m4a|aac|ts|mkv|webm|m4v|mov|ogg|flv|mpeg|mpg|3gp|avi)$/i.test(path);
    } catch(e) {
        return false;
    }
}



function cleanMediaTitle(value) {
    let name = String(value || '').trim();
    name = name.replace(/^(.*?)\s+-\s+\1(?=\s|$)/i, '$1').trim();
    // Dil/ülke etiketlerini yalnızca baştaki etiket olarak kaldır; TRT 1 gibi adlara dokunma.
    name = name.replace(/^(?:TR(?:\s*[:/.-]\s*|\s*🇹🇷\s*|\s+)|DE(?:\s*[:/.-]\s*|\s+|🇩🇪\s*))/iu, '');
    name = name.replace(/^(?:🇹🇷|🇩🇪)\s*/u, '');
    // Bölüm/sezon numaraları içerik başlığında tekrarlanmaz; oynatıcı alt bilgisinde gösterilir.
    name = name.replace(/\s*(?:[-–|·]\s*)?(?:S\s*\d{1,2}\s*E\s*\d{1,3}|Sezon\s*\d+\s*[-–:]?\s*\d+\s*\.?(?:bölüm|bolum)|\d+\s*\.?\s*(?:bölüm|bolum|episode))\s*$/i, '');
    return name.replace(/\s+/g, ' ').replace(/[\s|–-]+$/, '').trim();
}
function cleanGroupTitle(value) { return cleanMediaTitle(value); }
function cleanEpisodeTitle(value, fallbackNumber) {
    const name = cleanMediaTitle(value);
    const match = name.match(/(?:^|\s)(\d+)\s*\.?\s*(?:bölüm|bolum|episode)\s*$/i);
    return match ? (Number(match[1]) + '. Bölüm') : (fallbackNumber ? fallbackNumber + '. Bölüm' : name);
}



function parseM3U(text, prefix) {
    text = String(text || '').replace(/^\uFEFF/, '');

    const result = [];
    let current = null;
    let position = 0;
    let index = 0;

    while (position < text.length) {
        let end = text.indexOf('\n', position);
        if (end < 0) end = text.length;

        const line = text.slice(position, end).trim();
        position = end + 1;

        if (!line) continue;

        if (line.startsWith('#EXTINF:')) {
            const comma = line.lastIndexOf(',');
            const name = cleanMediaTitle(
                comma >= 0 ? line.slice(comma + 1) : 'İsimsiz yayın'
            );

            const group =
                (line.match(/group-title="([^"]*)"/i) || [])[1] || 'Genel';

            const logo =
                (line.match(/tvg-logo="([^"]*)"/i) || [])[1] || '';

            // Her EXTINF yeni ve bağımsız bir kanal başlatır.
            current = {
                name: name,
                group: group,
                logo: logo,
                url: '',
                headers: {},
                id: prefix + '-' + index
            };

            continue;
        }

        // Başlıklar yalnızca aktif kanal nesnesine yazılır.
        if (line.startsWith('#EXTVLCOPT:') && current) {
            const option = line.substring(11);
            const separator = option.indexOf('=');

            if (separator > 0) {
                const key = option.substring(0, separator).toLowerCase();
                const value = option.substring(separator + 1);

                if (key === 'http-user-agent') {
                    current.headers['User-Agent'] = value;
                } else if (
                    key === 'http-referrer' ||
                    key === 'http-referer'
                ) {
                    current.headers['Referer'] = value;
                } else if (key === 'http-origin') {
                    current.headers['Origin'] = value;
                }
            }

            continue;
        }

        if (line.startsWith('#EXTHTTP:') && current) {
            try {
                const extra = JSON.parse(line.substring(9));

                if (extra && typeof extra === 'object' && !Array.isArray(extra)) {
                    Object.keys(extra).forEach(key => {
                        const value = String(extra[key] == null ? '' : extra[key]);

                        if (
                            /^[a-z0-9-]+$/i.test(key) &&
                            !/^(host|content-length|connection|transfer-encoding|accept-encoding)$/i.test(key) &&
                            !/[\r\n]/.test(value)
                        ) {
                            current.headers[key] = value;
                        }
                    });
                }
            } catch (e) {}

            continue;
        }

        if (line.startsWith('#')) continue;

        if (current && isHttp(line)) {
            current.url = line;

            // Her kanalın kendi başlık nesnesinin kopyası saklanır.
            current.headers = Object.assign({}, current.headers);

            result.push(current);
            current = null;
            index++;
        }
    }

    return result;
}

function play(channel,queue=channels,index=-1,isLive=true) {
    if (!channel || (!isHttp(channel.url) && !channel.mag)) return;

    if (window.ISATV && window.ISATV.play) {
        window.ISATV.play(
            cleanMediaTitle(channel.name || 'İSA TV HD'),
            channel.url,
            JSON.stringify(channel.headers||{}),
            JSON.stringify((queue||[channel]).map(item => ({...item, is_live:item.is_live !== undefined ? item.is_live : !!isLive, name:cleanMediaTitle(item.name || 'İSA TV HD')}))),
            index < 0 ? Math.max(0,(queue||[]).indexOf(channel)) : index
        );
        return;
    }

    if (channel.mag) {
        alert('MAG yayını yerel oynatıcıda açılmalıdır.');
        return;
    }

    const query = new URLSearchParams({
        stream:channel.url,
        name:cleanMediaTitle(channel.name||'İSA TV HD'),
        group:channel.group||'',
        logo:channel.logo||''
    });

    location.href = './player.html?'+query.toString();
}

function playIsaFlix(name, url, headers={}, queue=[], index=0) {
    if (!isHttp(url)) {
        alert('Geçerli video bağlantısı bulunamadı.');
        return;
    }

    const list = Array.isArray(queue) && queue.length
        ? queue
        : [{
            name:name || 'İSAFLIX',
            url:url,
            headers:headers || {},
            season:0,
            episode:0,
            is_live:false
        }];

    let selected = Number(index);
    if (!Number.isInteger(selected) || selected < 0 || selected >= list.length) {
        selected = list.findIndex(x => x.url === url);
    }
    if (selected < 0) selected = 0;

    if (window.ISATV &&
        typeof window.ISATV.playIsaflix === 'function') {
        window.ISATV.playIsaflix(
            cleanMediaTitle(name || 'İSAFLIX'),
            url,
            JSON.stringify(headers || {}),
            JSON.stringify(list),
            selected
        );
        return;
    }

    play({
        name:name || 'İSAFLIX',
        url:url,
        logo:'',
        group:'İSAFLIX',
        headers:headers || {}
    }, list, selected, false);
}

function renderChannels() {
    const grid = document.getElementById('all-channels');
    if (!grid) return;

    grid.innerHTML = '';

    if (!channels.length) {
        grid.innerHTML = '<p class="isa-loading">Henüz kanal bulunamadı.</p>';
        renderFavorites();
        return;
    }

    const PAGE_SIZE = 500;
    const groupCounts = new Map();

    for (let i = 0; i < channels.length; i++) {
        const group = cleanGroupTitle(channels[i].group || 'Genel');
        groupCounts.set(group, (groupCounts.get(group) || 0) + 1);
    }

    const groupNodes = new Map();

    function getGroupGrid(group) {
        if (groupNodes.has(group)) return groupNodes.get(group);

        const details = document.createElement('details');
        details.className = 'isa-channel-category';

        const summary = document.createElement('summary');
        summary.textContent = group + ' (' + groupCounts.get(group) + ')';

        const list = document.createElement('div');
        list.className = 'channels-grid';

        details.appendChild(summary);
        details.appendChild(list);
        grid.appendChild(details);

        groupNodes.set(group, list);
        return list;
    }

    let shown = 0;
    let moreButton = null;

    function appendNextPage() {
        if (moreButton) {
            moreButton.remove();
            moreButton = null;
        }

        const end = Math.min(shown + PAGE_SIZE, channels.length);

        for (let index = shown; index < end; index++) {
            const channel = channels[index];
            const group = cleanGroupTitle(channel.group || 'Genel');
            const list = getGroupGrid(group);

            const card = document.createElement('div');
            card.className = 'channel-card';
            card.dataset.isaIndex = String(index);

            const thumbnail = document.createElement('div');
            thumbnail.className = 'channel-thumbnail';

            if (channel.logo) {
                const img = document.createElement('img');
                img.src = channel.logo;
                img.alt = '';
                img.onerror = function () {
                    this.style.display = 'none';
                };
                thumbnail.appendChild(img);
            } else {
                const placeholder = document.createElement('div');
                placeholder.className = 'channel-placeholder';
                placeholder.innerHTML = '<i class="fas fa-tv"></i>';
                thumbnail.appendChild(placeholder);
            }

            const name = document.createElement('div');
            name.className = 'channel-name';
            name.textContent = cleanMediaTitle(channel.name || 'İsimsiz kanal');

            card.appendChild(thumbnail);
            card.appendChild(name);

            card.addEventListener('click', function () {
                play(channels[index], channels, index);
            });

            list.appendChild(card);
        }

        shown = end;

        if (shown < channels.length) {
            moreButton = document.createElement('button');
            moreButton.type = 'button';
            moreButton.className = 'isa-load-more';
            moreButton.textContent =
                'Daha Fazla Kanal Yükle (' + shown + ' / ' +
                channels.length + ')';

            moreButton.style.display = 'block';
            moreButton.style.width = '100%';
            moreButton.style.padding = '14px';
            moreButton.style.margin = '16px 0';
            moreButton.style.cursor = 'pointer';

            moreButton.addEventListener('click', appendNextPage);
            grid.appendChild(moreButton);
        }
    }

    appendNextPage();
    renderFavorites();
}

async function loadUserLists(forceRefresh=false) {
    const result = [];

    for (let index=0; index<playlists.length; index++) {
        const item = playlists[index];

        try {
            if (item.type === 'xtream') {
                const live = await xtreamAPI(item, 'get_live_streams');

                const categories = await xtreamAPI(
                    item, 'get_live_categories'
                ).catch(() => []);

                const categoryMap = new Map(
                    (Array.isArray(categories) ? categories : []).map(c =>
                        [String(c.category_id), c.category_name || 'Genel']
                    )
                );

                for (const stream of (Array.isArray(live) ? live : [])) {
                    const ext = stream.container_extension || 'ts';
                    const url = item.host.replace(/\/+$/, '') +
                        '/live/' +
                        encodeURIComponent(item.username) + '/' +
                        encodeURIComponent(item.password) + '/' +
                        stream.stream_id + '.' + ext;

                    result.push({
                        name:stream.name || 'İsimsiz kanal',
                        url,
                        logo:stream.stream_icon || '',
                        group:categoryMap.get(
                            String(stream.category_id)
                        ) || 'Genel',
                        headers:{},
                        id:'xtream-live-' + index + '-' + stream.stream_id
                    });
                }

                item.lastCount = live.length;
                item.error = '';
                continue;
            }

            if (item.type === 'mag') {
                const magChannels = await loadMagChannels(item);

                for (const channel of magChannels) {
                    result.push(channel);
                }

                item.lastCount = magChannels.length;
                item.error = '';
                continue;
            }

            if (item.type === 'url' || item.type === 'm3u-url') {
                item.text = await getText(item.url, forceRefresh);
            }

            if (!item.text ||
                (!item.text.includes('#EXTM3U') &&
                 !item.text.includes('#EXTINF:'))) {
                throw new Error('Geçerli M3U içeriği bulunamadı.');
            }

            const parsed = parseM3U(
                item.text,
                'user-' + index
            );

            item.lastCount = parsed.length;
            item.error = '';

            // Dosya adı artık grup adına eklenmez.
            for (const channel of parsed) {
                channel.group = channel.group || 'Genel';
                result.push(channel);
            }

        } catch(error) {
            item.error = error.message || 'Kaynak yüklenemedi.';
            console.warn(item.name, error);
        }
    }

    saveLists();
    return result;
}



function xtreamAPI(source, action, extra={}) {
    const host = String(source.host || '').replace(/\/+$/, '');

    const params = new URLSearchParams({
        username:source.username,
        password:[REDACTED],
        action,
        ...extra
    });

    return getText(
        host + '/player_api.php?' + params.toString(),
        true
    ).then(text => {
        const data = JSON.parse(text);

        if (data && data.user_info &&
            data.user_info.auth === 0) {
            throw new Error('Xtream kullanıcı bilgileri geçersiz.');
        }

        if (data && data.user_info &&
            data.user_info.status === 'Disabled') {
            throw new Error('Xtream hesabı devre dışı.');
        }

        return data;
    });
}

function sourceHost(value) {
    let host = String(value || '').trim();

    if (!/^https?:\/\//i.test(host)) {
        host = 'http://' + host;
    }

    const parsed = new URL(host);

    if (!['http:', 'https:'].includes(parsed.protocol)) {
        throw new Error('Geçerli HTTP/HTTPS host girin.');
    }

    return {
        host:parsed.origin + parsed.pathname.replace(/\/+$/, ''),
        hasPort:!!parsed.port
    };
}

async function loadMagChannels(source) {
    const host = String(source.host || '').replace(/\/+$/, '');
    const mac = String(source.mac || '').trim().toUpperCase();

    if (!host || !/^([0-9A-F]{2}:){5}[0-9A-F]{2}$/.test(mac)) {
        throw new Error('MAG host veya MAC adresi geçersiz.');
    }

    const portal = host + '/portal.php';

    const baseHeaders = {
        'User-Agent':'Mozilla/5.0 (QtEmbedded; U; Linux; C) AppleWebKit/533.3',
        'X-User-Agent':'Model: MAG250; Link: WiFi',
        'Referer':host + '/c/',
        'Origin':new URL(host).origin,
        'Cookie':'mac=' + mac + '; stb_lang=en; timezone=Europe/Istanbul',
        'Accept':'*/*'
    };

    async function portalRequest(params, token=[REDACTED] {
        const query = new URLSearchParams({
            ...params,
            JsHttpRequest:'1-xml',
            mac:mac
        });

        const headers = {...baseHeaders};

        if (token) {
            headers.Authorization = 'Bearer ' + token;
        }

        const raw = await getTextWithHeaders(
            portal + '?' + query.toString(), headers);

        let response;

        try {
            response = JSON.parse(raw);
        } catch (e) {
            throw new Error('MAG portalı geçerli JSON döndürmedi.');
        }

        if (response && response.js && response.js.error) {
            throw new Error(String(response.js.error));
        }

        return response;
    }

    const handshake = await portalRequest({
        type:'stb',
        action:'handshake',
        token:[REDACTED]
    });

    const token = [REDACTED] && handshake.js && handshake.js.token
        ? String(handshake.js.token) : '';

    if (!token) {
        throw new Error('MAG handshake token alınamadı.');
    }

    try {
        await portalRequest({
            type:'stb',
            action:'get_profile'
        }, token);
    } catch (e) {}

    let genres = [];

    try {
        const genreResponse = await portalRequest({
            type:'itv',
            action:'get_genres'
        }, token);

        const genreData = genreResponse && genreResponse.js
            ? (genreResponse.js.data || genreResponse.js) : [];

        genres = Array.isArray(genreData) ? genreData : [];
    } catch (e) {}

    const genreMap = new Map();

    for (const genre of genres) {
        const id = genre.id ?? genre.genre_id ?? genre.tv_genre_id;
        const name = genre.title || genre.name || genre.genre_name;

        if (id != null && name) {
            genreMap.set(String(id), String(name));
        }
    }

    const channelResponse = await portalRequest({
        type:'itv',
        action:'get_all_channels',
        force_ch_link_check:''
    }, token);

    let rows = channelResponse && channelResponse.js
        ? channelResponse.js.data : [];

    if (rows && !Array.isArray(rows) && Array.isArray(rows.data)) {
        rows = rows.data;
    }

    if (!Array.isArray(rows) || !rows.length) {
        throw new Error('MAG portalı kanal listesi döndürmedi.');
    }

    const result = [];
    const MAX_MAG_CHANNELS = Number.MAX_SAFE_INTEGER;

    for (let index = 0; index < rows.length; index++) {
        if (result.length >= MAX_MAG_CHANNELS) break;

        const row = rows[index] || {};

        const genreId = row.tv_genre_id ??
                        row.genre_id ??
                        row.category_id ?? '';

        const groupName = String(
            row.tv_genre_name ||
            genreMap.get(String(genreId)) ||
            ''
        ).trim();

        // Yalnızca grup adında TR geçen kanallar.
        if (!groupName.toLocaleUpperCase('tr-TR').includes('TR')) {
            continue;
        }

        const cmd = String(row.cmd || row.url || '').trim();

        if (!cmd) continue;

        const direct = cmd
            .replace(/^(ffmpeg|auto)\s+/i, '')
            .trim();

        result.push({
            name:row.name || row.title || ('MAG Kanal ' + (index + 1)),
            url:/^https?:\/\//i.test(direct) ? direct : '',
            logo:row.logo || row.icon || '',
            group:groupName,
            headers:{
                ...baseHeaders,
                Authorization:'Bearer ' + token
            },
            mag:true,
            mag_host:host,
            mag_mac:mac,
            mag_token:[REDACTED],
            mag_cmd:cmd,
            id:'mag-' + mac.replace(/:/g,'') + '-' + index
        });
    }

    if (!result.length) {
        throw new Error(
            'MAG kaynağında grup adında TR geçen kanal bulunamadı.');
    }

    return result;
}

async function detectXtreamM3U(url) {
    try {
        const parsed = new URL(url);
        const username = parsed.searchParams.get('username');
        const password = [REDACTED];

        if (!username || !password) return null;

        const source = {
            host:parsed.origin,
            username,
            password
        };

        const data = await xtreamAPI(
            source,
            'get_live_categories'
        );

        if (!Array.isArray(data)) return null;

        return source;
    } catch(error) {
        return null;
    }
}

function normalizeVOD(data,result=[]) {
    if (Array.isArray(data)) {
        data.forEach(item => normalizeVOD(item,result));
        return result;
    }

    if (!data || typeof data !== 'object') return result;

    const title = data.title || data.name || data.baslik ||
                  data.titleName || data.filename || data.original_title;

    const url = data.url || data.link || data.file ||
                data.stream || data.playUrl || data.video ||
                data.videoUrl || data.video_url || data.embedUrl || '';

    const episodes = Array.isArray(data.episodes) ? data.episodes : [];
    const seasons = Array.isArray(data.seasons) ? data.seasons : [];
    const items = Array.isArray(data.items) ? data.items : [];

    if (title && (url || episodes.length || seasons.length || items.length ||
                  String(data.type || '').toLowerCase().includes('dizi') ||
                  String(data.type || '').toLowerCase().includes('series'))) {
        result.push({
            type:String(data.type || '').toLowerCase(),
            name:String(title),
            title:String(title),
            url:url,
            videoUrl:data.videoUrl || data.video_url || '',
            logo:data.poster || data.image || data.cover ||
                 data.thumbnail || data.thumb || '',
            poster:data.poster || data.image || data.cover ||
                   data.thumbnail || '',
            description:data.description || '',
            category:data.category || '',
            genres:Array.isArray(data.genres) ? data.genres : [],
            year:data.year || '',
            platform:data.platform || '',
            episodes:episodes,
            seasons:seasons,
            items:items,
            raw:data
        });
        return result;
    }

    Object.values(data).forEach(value => {
        if (value && typeof value === 'object') normalizeVOD(value,result);
    });

    return result;
}

    for (let i = 0; i < items.length; i++) {
        const entry = {item:items[i], index:i};

        else films.push(entry);
    }

    area.innerHTML = `
        <div class="isa-vod-toolbar">
            <button class="isa-back" id="isa-vod-back">
                <i class="fas fa-arrow-left"></i> Kaynaklar
            </button>
            <h2>${esc(providerName(provider))}</h2>
        </div>
        ${series.length ? '<h3>Diziler</h3><div id="isa-series-grid" class="isa-provider-grid"></div>' : ''}
        ${films.length ? '<h3 style="margin-top:25px">Filmler ve diğer içerikler</h3><div id="isa-films-grid" class="isa-provider-grid"></div>' : ''}
        ${!items.length ? '<p class="isa-vod-note">Bu kaynakta gösterilebilir içerik bulunamadı.</p>' : ''}
    `;

    document.getElementById('isa-vod-back').onclick = showVODProviders;

    const renderBatch = (entries, gridId) => {
        const grid = document.getElementById(gridId);
        if (!grid) return;

        let offset = 0;
        const batchSize = 100;

        const next = () => {
            if (!grid.isConnected) return;

            const fragment = document.createDocumentFragment();
            const limit = Math.min(offset + batchSize, entries.length);

            for (; offset < limit; offset++) {
                const entry = entries[offset];
                const item = entry.item;
                const kind = itemIsSeries(item) ? 'series' : 'film';

                const holder = document.createElement('div');
                holder.innerHTML = vodCard(item,kind,entry.index);

                const button = holder.firstElementChild;
                button.addEventListener('click', () => {
                    if (kind === 'series') {
                        if (provider.type === 'xtream' && item.xtream)
                            openXtreamSeries(item,provider);
                        else
                            showSeries(item,provider);
                    } else {
                        openVODItem(item);
                    }
                });

                fragment.appendChild(button);
            }

            grid.appendChild(fragment);

            if (offset < entries.length) {
                requestAnimationFrame(next);
            }
        };

        requestAnimationFrame(next);
    };

    renderBatch(series,'isa-series-grid');
    renderBatch(films,'isa-films-grid');
}

async function openVODProvider(provider) {
    const area = document.getElementById('isa-vod-items');
    const grid = document.getElementById('isa-vod-grid');
    const heading = document.querySelector('#isa-vod-panel .isa-section-heading');

    currentProvider = provider;
    if (grid) grid.style.display = 'none';
    if (heading) heading.style.display = 'none';
    area.innerHTML = '<p class="isa-loading">İçerikler yükleniyor...</p>';

    try {
        let items = [];
        if (provider.type === 'xtream') {
            const [films,series,filmCategories,seriesCategories] = await Promise.all([
                xtreamAPI(provider,'get_vod_streams'),
                xtreamAPI(provider,'get_series'),
                xtreamAPI(provider,'get_vod_categories').catch(() => []),
                xtreamAPI(provider,'get_series_categories').catch(() => [])
            ]);

            const filmMap = new Map();
            for (const category of (Array.isArray(filmCategories) ? filmCategories : [])) {
                filmMap.set(String(category.category_id),category.category_name || 'Filmler');
            }
            const seriesMap = new Map();
            for (const category of (Array.isArray(seriesCategories) ? seriesCategories : [])) {
                seriesMap.set(String(category.category_id),category.category_name || 'Diziler');
            }

            for (const item of (Array.isArray(films) ? films : [])) {
                items.push({
                    ...item,
                    type:'film',
                    name:item.name || 'Film',
                    title:item.name || 'Film',
                    category:filmMap.get(String(item.category_id)) || 'Filmler',
                    url:provider.host.replace(/\/+$/,'') +
                        '/movie/' + encodeURIComponent(provider.username) + '/' +
                        encodeURIComponent(provider.password) + '/' +
                        item.stream_id + '.' + (item.container_extension || 'mp4'),
                    logo:item.stream_icon || item.cover || '',
                    raw:item
                });
            }

            for (const item of (Array.isArray(series) ? series : [])) {
                items.push({
                    ...item,
                    type:'series',
                    name:item.name || 'Dizi',
                    title:item.name || 'Dizi',
                    category:seriesMap.get(String(item.category_id)) || 'Diziler',
                    url:'',
                    logo:item.cover || item.stream_icon || '',
                    raw:item,
                    xtream:true
                });
            }
        } else {
            const text = await getText(provider.url);
            if (text.trim().startsWith('#EXTM3U') || text.includes('#EXTINF:')) {
                items = parseM3U(text,'vod').map(item => ({
                    ...item,title:item.name,logo:item.logo,episodes:[],seasons:[],type:'film'
                }));
            } else {
                items = normalizeVOD(JSON.parse(text));
            }
        }
        renderVODItems(items,provider);
    } catch(error) {
        area.innerHTML = `
            <div class="isa-vod-toolbar">
                <button class="isa-back" id="isa-vod-error-back">Geri</button>
                <h2>${esc(providerName(provider))}</h2>
            </div>
            <p class="isa-vod-note">İçerik alınamadı: ${esc(error.message)}</p>
        `;
        const back = document.getElementById('isa-vod-error-back');
        if (back) back.onclick = showVODProviders;
    }
}

async function openXtreamSeries(series,provider) {
    const area = document.getElementById('isa-vod-items');
    area.innerHTML = '<p class="isa-loading">Dizi bölümleri alınıyor...</p>';

    try {
        const info = await xtreamAPI(provider, 'get_series_info', {
            series_id:series.series_id
        });

        const rawEpisodes = info.episodes || {};
        const seasons = info.seasons || [];
        const episodes = [];

        if (Array.isArray(rawEpisodes)) {
            for (const episode of rawEpisodes) episodes.push(episode);
        } else {
            Object.values(rawEpisodes).forEach(list => {
                if (Array.isArray(list)) { for (const episode of list) episodes.push(episode); }
            });
        }

        const grouped = new Map();

        for (const episode of episodes) {
            const season = Number(
                episode.season || episode.season_number || 1
            );

            if (!grouped.has(season)) grouped.set(season, []);

            const ext = episode.container_extension || 'mp4';

            grouped.get(season).push({
                ...episode,
                name:episode.title || episode.name || 'Bölüm',
                title:episode.title || episode.name || 'Bölüm',
                url:provider.host.replace(/\/+$/, '') +
                    '/series/' +
                    encodeURIComponent(provider.username) + '/' +
                    encodeURIComponent(provider.password) + '/' +
                    episode.id + '.' + ext
            });
        }

        const seasonNumbers = Array.from(grouped.keys()).sort((a,b) => a-b);

        area.innerHTML = `
            <div class="isa-vod-toolbar">
                <button class="isa-back" id="isa-xtream-series-back">
                    <i class="fas fa-arrow-left"></i> Geri
                </button>
                <h2>${esc(series.name)}</h2>
            </div>
            <div class="isa-provider-grid">
                ${seasonNumbers.map((number,index) => `
                    <button class="isa-provider-card"
                            data-xseason="${index}">
                        <i class="fas fa-layer-group"
                           style="font-size:32px;padding:20px"></i>
                        <strong>Sezon ${number}</strong>
                        <small>${grouped.get(number).length} bölüm</small>
                    </button>
                `).join('')}
            </div>
        `;

        document.getElementById('isa-xtream-series-back').onclick =
            () => openVODProvider(provider);

        area.querySelectorAll('[data-xseason]').forEach(button => {
            button.onclick = () => showEpisodes(
                grouped.get(seasonNumbers[
                    Number(button.dataset.xseason)
                ]),
                series,
                provider
            );
        });

        if (!seasonNumbers.length) {
            area.innerHTML +=
                '<p class="isa-vod-note">Bu dizide bölüm bulunamadı.</p>';
        }

    } catch(error) {
        area.innerHTML =
            '<p class="isa-vod-note">Dizi bilgileri alınamadı: ' +
            esc(error.message) + '</p>';
    }
}

function seasonNumber(item) {
    const explicit = item.season_number ?? item.season;
    if (explicit !== undefined && explicit !== null &&
        Number.isFinite(Number(explicit))) {
        return Number(explicit);
    }

    const title = String(item.title || item.name || '');
    const patterns = [
        /(?:^|\b)(\d+)\s*\.?\s*(?:sezon|season)\b/i,
        /(?:season|sezon)\s*0*(\d+)/i,
        /\bS0*(\d+)E0*\d+\b/i,
        /\b0*(\d+)x0*\d+\b/i
    ];

    for (const pattern of patterns) {
        const match = title.match(pattern);
        if (match) return Number(match[1]);
    }
    return 1;
}

async function resolveStaticMedia(page, depth=0, visited=new Set()) {
    if (!isHttp(page) || depth > 2 || visited.has(page)) return null;
    visited.add(page);

    let html;
    try {
        html = await nativeFetch(page);
    } catch (e) {
        console.warn('VOD sayfası okunamadı:', e);
        return null;
    }

    if (!html) return null;

    const patterns = [
        /<source[^>]+src=["']([^"']+\.(?:m3u8|mp4|mkv|webm)(?:\?[^"']*)?)["']/ig,
        /<video[^>]+src=["']([^"']+\.(?:m3u8|mp4|mkv|webm)(?:\?[^"']*)?)["']/ig,
        /(?:file|source|src|hls|playlist)\s*[:=]\s*["']([^"']+\.(?:m3u8|mp4|mkv|webm)(?:\?[^"']*)?)["']/ig,
        /<meta[^>]+property=["']og:video["'][^>]+content=["']([^"']+)["']/ig
    ];

    for (const pattern of patterns) {
        let match;
        while ((match = pattern.exec(html)) !== null) {
            try {
                const candidate = new URL(
                    match[1].replace(/&amp;/g,'&'),
                    page
                ).href;

                if (isDirectMedia(candidate)) {
                    return candidate;
                }
            } catch (e) {}
        }
    }

    // Yalnızca sayfada açıkça belirtilmiş iframe adreslerini kontrol et.
    const iframePattern = /<iframe[^>]+src=["']([^"']+)["']/ig;
    let iframe;

    while ((iframe = iframePattern.exec(html)) !== null) {
        try {
            const child = new URL(iframe[1],page).href;
            if (!isHttp(child)) continue;

            const found = await resolveStaticMedia(
                child,
                depth+1,
                visited
            );

            if (found) return found;
        } catch (e) {}
    }

    return null;
}

async function openVODItem(item) {
    const candidates = [
        item.videoUrl,
        item.video_url,
        item.playUrl,
        item.stream,
        item.file,
        item.url,
        item.link,
        item.embedUrl
    ].filter(Boolean);

    const media = candidates.find(isDirectMedia);

    if (media) {
        play({
            name:item.title||item.name||'İSA Flix',
            url:media,
            logo:item.poster||item.logo||'',
            group:'İSA Flix',
            headers:item.headers||{}
        },[],0);
        return;
    }

    const page = candidates.find(isHttp);

    if (!page) {
        alert('Bu içerik için geçerli bir bağlantı bulunamadı.');
        return;
    }

    const area = document.getElementById('isa-vod-items');
    const note = document.createElement('p');
    note.className = 'isa-vod-note';
    note.textContent = 'Video kaynağı aranıyor...';
    area.prepend(note);

    try {
        const resolved = await resolveStaticMedia(page);

        note.remove();

        if (!resolved) {
            alert(
                'Bu kaynağın sayfasında doğrudan ve erişilebilir ' +
                'bir video adresi bulunamadı. Kaynak dinamik oynatıcı ' +
                'veya erişim koruması kullanıyor olabilir.'
            );
            return;
        }

        play({
            name:item.title||item.name||'İSA Flix',
            url:resolved,
            logo:item.poster||item.logo||'',
            group:'İSA Flix',
            headers:{
                ...(item.headers||{}),
                Referer:page
            }
        },[],0);

    } catch (error) {
        note.remove();
        console.error('VOD çözümleme hatası:',error);
        alert('Video kaynağı çözümlenemedi: '+error.message);
    }
}


async function openIsaFlixVODItem(item) {

async function addSource() {
    const type = document.getElementById('isa-source-type').value;
    if (!type) {
        alert('Önce kaynak türünü seçin.');
        return;
    }

    sourceLoading(true, 'Kaynak kontrol ediliyor...');

    try {
        if (type === 'm3u') {
            const url = document.getElementById(
                'isa-new-m3u-url'
            ).value.trim();

            if (!isHttp(url)) {
                throw new Error('Geçerli HTTP/HTTPS M3U adresi girin.');
            }

            await addM3UUrl(url);

        } else if (type === 'xtream') {
            const hostInput = document.getElementById('isa-xt-host').value;
            const user = document.getElementById('isa-xt-user').value.trim();
            const pass = document.getElementById('isa-xt-pass').value;

            if (!hostInput.trim() || !user || !pass) {
                throw new Error('Host, kullanıcı adı ve şifre zorunludur.');
            }

            let parsed = sourceHost(hostInput);

            if (!parsed.hasPort) {
                const port = document.getElementById('isa-xt-port').value.trim();
                if (!port || !/^\d{1,5}$/.test(port) ||
                    Number(port) < 1 || Number(port) > 65535) {
                    throw new Error('Geçerli bir port girin.');
                }

                parsed.host = parsed.host.replace(/\/+$/, '') +
                    ':' + port;
            }

            await addXtreamSource(parsed.host,user,pass);

        } else if (type === 'mag') {
            const host = sourceHost(
                document.getElementById('isa-mag-host').value
            ).host;

            const mac = document.getElementById(
                'isa-mag-mac'
            ).value.trim().toUpperCase();

            if (!/^([0-9A-F]{2}:){5}[0-9A-F]{2}$/.test(mac)) {
                throw new Error('MAC adresini 00:1A:79:XX:XX:XX biçiminde girin.');
            }

            const source = {
                host,
                mac,
                type:'mag',
                name:nextSourceName('mag')
            };

            // Kanal listesi refreshAll içinde yalnızca bir kez yüklenecek.
            playlists.push(source);
            saveLists();
            await refreshAll(false);
        }

        document.getElementById('isa-upload-status').textContent =
            'Kaynak başarıyla eklendi.';

        document.getElementById('isa-source-type').value = '';
        renderSourceFields();
        const form=document.getElementById('isa-source-form'); if(form) form.hidden=true;
        const toggle=document.getElementById('isa-toggle-source-form'); if(toggle) toggle.textContent='＋ Liste ekle';

    } catch(error) {
        document.getElementById('isa-upload-status').textContent =
            error.message || 'Kaynak eklenemedi.';
        console.error(error);
    } finally {
        sourceLoading(false);
    }
}

async function openMediaSource(source,kind,root) {
    localStorage.setItem('isa-active-media-'+kind,mediaSourceKey(source));
    const cacheKey=mediaSourceKey(source);
    if (isaMediaMemory[kind].has(cacheKey)) { renderMediaGroups(isaMediaMemory[kind].get(cacheKey),source,kind,root); return; }
    root.innerHTML = '<p class="isa-loading">Kaynak içerikleri yükleniyor...</p>';
    try {
        let items = [], categories = new Map();
        if (source.type === 'xtream') {
            const action = kind === 'film' ? 'get_vod_streams' : 'get_series';
            const catAction = kind === 'film' ? 'get_vod_categories' : 'get_series_categories';
            const [data,cats] = await Promise.all([xtreamAPI(source,action),xtreamAPI(source,catAction).catch(()=>[])]);
            for (const c of (Array.isArray(cats)?cats:[])) categories.set(String(c.category_id),c.category_name || 'Genel');
            items = (Array.isArray(data)?data:[]).map(x => ({
                name:x.name || (kind==='film'?'Film':'Dizi'),
                logo:x.stream_icon || x.cover || '',
                category:categories.get(String(x.category_id)) || 'Genel',
                raw:x, source, kind,
                url:kind==='film' ? source.host.replace(/\/+$/,'')+'/movie/'+encodeURIComponent(source.username)+'/'+encodeURIComponent(source.password)+'/'+x.stream_id+'.'+(x.container_extension||'mp4') : ''
            }));
        } else {
            let text = source.text;
            if (!text && source.url) text = await getText(source.url,true);
            const parsed = parseM3U(text || '', 'media');
            items = parsed.filter(x => {
                const g=String(x.group||'').toLocaleLowerCase('tr');
                const n=String(x.name||'').toLocaleLowerCase('tr');
                return kind==='film' ? /film|movie|sinema/.test(g+' '+n) : /dizi|series|season|sezon|episode|bölüm/.test(g+' '+n);
            }).map(x=>({...x,category:x.group||'Genel',kind,source}));
        }
        const groups = new Map();
        for (const item of items) { const c=cleanGroupTitle(item.category||'Genel'); item.category=c; if (!groups.has(c)) groups.set(c,[]); groups.get(c).push(item); }
        isaMediaMemory[kind].set(cacheKey,groups);
        renderMediaGroups(groups,source,kind,root);
    } catch(e) {
        root.innerHTML=`<button class="isa-back" data-media-back>Tekrar dene</button><p class="isa-vod-note">Kaynak yüklenemedi: ${esc(e.message)}</p>`;
        root.querySelector('[data-media-back]').onclick=()=>renderMediaPanel(kind);
    }
}

const isaSeriesMetaCache = new Map();

async function isaGetSeriesMeta(item) {
    const id = item.raw && item.raw.series_id;
    if (!id) return null;
    const key = [item.source.host,item.source.username,id].join('|');
    if (isaSeriesMetaCache.has(key)) return isaSeriesMetaCache.get(key);
    const info = await xtreamAPI(item.source,'get_series_info',{series_id:id});
    const eps = info && info.episodes ? info.episodes : {};
    const seasonKeys = Array.isArray(eps) ? ['1'] : Object.keys(eps);
    let total = 0, maxSeason = 0;
    for (const k of seasonKeys) {
        const list = Array.isArray(eps) ? eps : (Array.isArray(eps[k]) ? eps[k] : []);
        total += list.length;
        const n = Number(k);
        if (Number.isFinite(n)) maxSeason = Math.max(maxSeason,n);
        for (const ep of list) maxSeason = Math.max(maxSeason,Number(ep.season||ep.season_number||0)||0);
    }
    const seasons = Array.isArray(info.seasons) ? info.seasons : [];
    for (const season of seasons) maxSeason = Math.max(maxSeason,Number(season.season_number||season.season||0)||0);
    const result = {maxSeason,total};
    isaSeriesMetaCache.set(key,result);
    return result;
}

async function isaFillSeriesMeta(items,root) {
    const cards = Array.from(root.querySelectorAll('[data-media-item]'));
    let cursor = 0;
    const workers = Array.from({length:Math.min(3,items.length)},async()=>{
        while(cursor < items.length) {
            const i=cursor++;
            const item=items[i], card=cards[i];
            if (!card || !item || !item.raw || !item.raw.series_id) continue;
            const small=card.querySelector('small');
            if (small) small.textContent='Dizi bilgileri alınıyor...';
            try {
                const meta=await isaGetSeriesMeta(item);
                if (small && card.isConnected && meta) {
                    const seasonText=meta.maxSeason ? `Sezon ${meta.maxSeason}` : 'Sezon bilgisi yok';
                    const strong=card.querySelector('strong');
                    if (strong) strong.textContent=cleanMediaTitle(item.name);

async function openMediaSeries(series,source,root,kind,category) {
    root.innerHTML='<p class="isa-loading">Sezonlar yükleniyor...</p>';

        if(Array.isArray(raw)) { for(const ep of raw) all.push(ep); }
        else for(const key of Object.keys(raw)) if(Array.isArray(raw[key])) for(const ep of raw[key]) all.push(ep);
        const seasons=new Map();
        for(const ep of all) {
            const sn=Number(ep.season||ep.season_number||1);
            if(!seasons.has(sn)) seasons.set(sn,[]);

function initScrollArrows() {
    if (document.getElementById('isa-scroll-arrows')) return;
    const box = document.createElement('div');
    box.id = 'isa-scroll-arrows';
    box.innerHTML = '<button type="button" aria-label="Yukarı çık">▲</button><button type="button" aria-label="Sayfanın sonuna in">▼</button>';
    document.body.appendChild(box);
    const buttons = box.querySelectorAll('button');
    buttons[0].onclick = () => window.scrollTo({top:0,behavior:'smooth'});
    buttons[1].onclick = () => window.scrollTo({top:document.documentElement.scrollHeight,behavior:'smooth'});
    let lastY = window.scrollY, lastT = performance.now(), hideTimer = 0;
    window.addEventListener('scroll', () => {
        const now = performance.now(), y = window.scrollY;
        const speed = Math.abs(y-lastY) / Math.max(1,now-lastT);
        lastY=y; lastT=now;
        const canScroll = document.documentElement.scrollHeight > window.innerHeight + 120;
        if (canScroll && speed > 0.22) {
            box.classList.add('visible');
            buttons[0].style.visibility = y > 180 ? 'visible' : 'hidden';
            buttons[1].style.visibility = y < document.documentElement.scrollHeight-window.innerHeight-100 ? 'visible' : 'hidden';
            clearTimeout(hideTimer); hideTimer=setTimeout(()=>box.classList.remove('visible'),950);
        } else if (speed <= 0.22) {
            clearTimeout(hideTimer); hideTimer=setTimeout(()=>box.classList.remove('visible'),500);
        }
    },{passive:true});
}

function isaFlixArray(value) {
    return Array.isArray(value) ? value : [];
}

function isaFlixText(value, fallback='') {
    return String(value == null ? fallback : value);
}

function isaFlixHeaders(item) {
    const headers = {...(item.headers || {})};
    if (item.referer) headers.Referer = item.referer;
    if (item.user_agent) headers['User-Agent'] = item.user_agent;
    return headers;
}

function isaFlixModeItems(data, mode) {
    if (mode === 'live') return isaFlixArray(data.live_channels);
    if (mode === 'movies') return isaFlixArray(data.movies);
    return isaFlixArray(data.series);

function isaFlixCategory(item, mode) {
    if (mode === 'live') return isaFlixText(item.category_name, 'Genel');
    if (mode === 'movies') return isaFlixText(item.category_name, 'Diğer');
    return isaFlixText(item.platform_name || item.category_name, 'Diğer');
}

function isaFlixCategories(data, mode) {
    let source = [];
    if (mode === 'live') source = isaFlixArray(data.live_categories);
    if (mode === 'movies') source = isaFlixArray(data.movie_categories);
    if (mode === 'series') source = isaFlixArray(data.series_platforms);

    let names = source.map(x => isaFlixText(x.name || x.category_name).trim()).filter(Boolean);
    if (!names.length) {
        names = [...new Set(isaFlixModeItems(data, mode).map(x =>
            isaFlixCategory(x, mode)
        ))];
    }
    return [...new Set(names)];
}

function isaFlixRenderItems() {
    const data = isaFlixCatalog();
    const grid = document.getElementById('isaf-grid');
    const status = document.getElementById('isaf-status');
    const more = document.getElementById('isaf-more');
    if (!grid || !status || !more) return;

    const mode = isaFlixState.mode;
    const query = isaFlixState.query.trim().toLocaleLowerCase('tr');
    const category = isaFlixState.category;

    let items = isaFlixModeItems(data, mode);

    items = items.filter(item => {
         if (category && isaFlixCategory(item, mode) !== category) return false;
        if (!query) return true;

        const text = [
            item.name, item.title, item.category_name,
            item.platform_name, item.description, item.year
        ].join(' ').toLocaleLowerCase('tr');

        return text.includes(query);
    });

    isaFlixState.items = items;
    isaFlixState.shown = 0;
    grid.innerHTML = '';

    status.textContent = isaFlixTvLoading
        ? 'İçerikler indiriliyor…'
        : items.length.toLocaleString('tr-TR') + ' içerik bulundu.';

    if (!items.length) {
        const message = isaFlixTvLoading
            ? 'İçerikler indiriliyor…'
            : 'Burada bir şey yok';

        grid.innerHTML = '<p class="isa-vod-note"></p>';
        grid.firstElementChild.textContent = message;
        more.hidden = true;
        return;
    }

    more.hidden = false;
    more.onclick = isaFlixLoadMore;
    isaFlixLoadMore();
}

        const title = item.title || item.name || 'İsimsiz içerik';
        const image = item.poster_url || item.logo_url ||
            item.image_url || item.poster || item.image || '';

            : [item.year, isaFlixCategory(item,mode)].filter(Boolean).join(' · ');

        card.type = 'button';
        card.className = 'isaf-card';

        img.loading = 'lazy';
        img.decoding = 'async';

        img.onerror = () => { img.style.display = 'none'; };

        const name = document.createElement('strong');

function isaFlixOpenItem(item) {
    const mode = isaFlixState.mode;
    const currentItems = Array.isArray(isaFlixState.items)
        ? isaFlixState.items : [];
    const currentIndex = currentItems.indexOf(item);

    if (mode === 'live') {
        if (!item || !item.id) {
            alert('Kanal bilgisi bulunamadı.');
            return;
        }
        isaFlixNativeCall(
            'memberGetTvPlayback',
            [String(item.id)],
            '__isaTvPlaybackDone',
            result => {
                if (!isaFlixAuthenticated) return;
                const url = String(result?.playback_url || '');
                if (!result?.ok || !/^https?:\/\//i.test(url)) {
                    alert(result?.error || 'Kanal başlatılamadı.');
                    return;
                }
                const queue = currentItems.map(x => ({
                    name: x.name || x.title || 'Kanal',
                    url: '',
                    logo: x.logo || x.logo_url || '',
                    group: isaFlixCategory(x, 'live'),
                    headers: {},
                    is_live: true,
                    playback_type: 'live',
                    channel_id: String(x.id || '')
                })).filter(x => x.channel_id);

                const selected = queue.findIndex(
                    x => x.channel_id === String(item.id)
                );
                const channel = queue[selected >= 0 ? selected : 0];
                if (!channel) {
                    alert('Kanal listesi hazırlanamadı.');
                    return;
                }
                channel.url = url;
                playIsaFlix(channel.name, url, {}, queue,
                    selected >= 0 ? selected : 0);
            }
        );
        return;
    }

    if (mode === 'movies') {
        if (!item || !item.id) {
            alert('Film bilgisi bulunamadı.');
            return;
        }

        isaFlixNativeCall(
            'memberGetContentPlayback',
            ['movie', String(item.id), ''],
            '__isaContentPlaybackDone',
            result => {
                if (!isaFlixAuthenticated) return;
                const url = String(result?.playback_url || '');
                if (!result?.ok || !/^https?:\/\//i.test(url)) {
                    alert(result?.error || 'Film başlatılamadı.');

                const name = cleanMediaTitle(item.title || item.name || 'Film');

                    name: cleanMediaTitle(x.title || x.name || 'Film'),

                    logo: x.poster_url || x.image_url || '',
                    group: isaFlixCategory(x, 'movies'),

                    queue, queueIndex: selected >= 0 ? selected : 0
                });
            }
        );
        return;
    }

    isaFlixShowSeries(item);
}

function isaFlixExplicitSeasonNumber(item) {
    const raw = item.season_number ?? item.season;

    if (raw !== undefined && raw !== null &&
        String(raw).trim() !== '' &&
        Number.isFinite(Number(raw)) && Number(raw) > 0) {
        return Number(raw);
    }

    const hint = [
        item.title, item.name, item.page_url, item.url,
        item.season
    ].filter(Boolean).join(' ');

    const patterns = [
        /\bS0*(\d+)E0*\d+\b/i,
        /\b0*(\d+)x0*\d+\b/i,
        /\b(?:season|sezon)[\s._-]*0*(\d+)\b/i,
        /\b0*(\d+)[\s._-]*(?:season|sezon)\b/i
    ];

    for (const pattern of patterns) {
        const match = hint.match(pattern);
        if (match) return Number(match[1]);
    }

    return 0;
}

function isaFlixExplicitEpisodeNumber(item) {
    const raw = item.episode_number ?? item.episode ?? item.episode_num;

    if (raw !== undefined && raw !== null &&
        String(raw).trim() !== '' &&
        Number.isFinite(Number(raw)) && Number(raw) > 0) {
        return Number(raw);
    }

    const hint = [
        item.title, item.name, item.page_url, item.url
    ].filter(Boolean).join(' ');

    const patterns = [
        /\bS\d+E0*(\d+)\b/i,
        /\b\d+x0*(\d+)\b/i,
        /(?:bölüm|bolum|episode|ep)[\s._-]*0*(\d+)/i
    ];

    for (const pattern of patterns) {
        const match = hint.match(pattern);
        if (match) return Number(match[1]);
    }

    return 0;
}


async function isaFlixShowSeries(series) {
    const root = isaFlixRoot();
    if (!root || !series) return;

    const seriesId = String(series.series_id || series.id || '');
    if (!seriesId) {
        alert('Dizi bilgisi bulunamadı.');
        return;
    }

    root.innerHTML = `
        <div class="isaf-detail">
            <button class="isa-back" id="isaf-back">← Dizilere dön</button>
            <h2>${esc(series.title || series.name || 'Dizi')}</h2>
            <p class="isa-vod-note">Bölümler yükleniyor...</p>
        </div>`;
    const back = document.getElementById('isaf-back');
    if (back) back.onclick = () => isaFlixRender();

    try {
        const result = await isaFlixFetchTvSeries(seriesId);

        if (result?.authenticated === false) {
            isaFlixAuthenticated = false;
            isaFlixShowGate('Oturum süresi doldu. Yeniden giriş yapın.');
            return;
        }

        if (!result?.ok || !Array.isArray(result.episodes)) {
            const currentRoot = isaFlixRoot();
            if (currentRoot) {
                currentRoot.innerHTML = `
                    <div class="isaf-detail">
                        <button class="isa-back" id="isaf-back">← Dizilere dön</button>
                        <h2>${esc(series.title || series.name || 'Dizi')}</h2>
                        <p class="isa-vod-note">${
                            esc(result?.error || 'Bölüm listesi alınamadı.')
                        }</p>
                    </div>`;
                const button = document.getElementById('isaf-back');
                if (button) button.onclick = () => isaFlixRender();
            }
            return;
        }

        const episodeSource = result.episodes;
        const episodes = Array.isArray(episodeSource)
            ? episodeSource
            : (episodeSource && typeof episodeSource === 'object'
                ? Object.entries(episodeSource).flatMap(([season, list]) =>
                    Array.isArray(list)
                        ? list.map(ep => ({
                            ...ep,
                            season: ep.season ?? (Number(season) || 1)
                        }))
                        : []
                  )
                : []);
        const prepared = episodes.map((episode, index) => ({
            ...episode,
            __index:index,
            __realSeason:isaFlixExplicitSeasonNumber(episode),
            __realEpisode:isaFlixExplicitEpisodeNumber(episode)
        }));

        // Sezon kartı oluşturulmaz; bütün bölümler tek listede sıralanır.
        prepared.sort((a, b) => {
            const sa = a.__realSeason || 0;
            const sb = b.__realSeason || 0;
            if (sa !== sb) return sa - sb;

            const ea = a.__realEpisode || 0;
            const eb = b.__realEpisode || 0;
            if (ea && eb && ea !== eb) return ea - eb;
            if (ea && !eb) return -1;
            if (!ea && eb) return 1;
            return a.__index - b.__index;
        });

        if (!prepared.length) {
            const currentRoot = isaFlixRoot();
            if (currentRoot) {
                currentRoot.innerHTML = `
                    <div class="isaf-detail">
                        <button class="isa-back" id="isaf-back">← Dizilere dön</button>
                        <h2>${esc(series.title || series.name || 'Dizi')}</h2>
                        <p class="isa-vod-note">Bu dizide bölüm bulunamadı.</p>
                    </div>`;
                const button = document.getElementById('isaf-back');
                if (button) button.onclick = () => isaFlixRender();
            }
            return;
        }

        isaFlixShowEpisodes(series, 0, prepared);
    } catch (error) {
        console.error('series-episodes-load-failed');
        const currentRoot = isaFlixRoot();
        if (currentRoot) {
            currentRoot.innerHTML = `
                <div class="isaf-detail">
                    <button class="isa-back" id="isaf-back">← Dizilere dön</button>
                    <h2>${esc(series.title || series.name || 'Dizi')}</h2>
                    <p class="isa-vod-note">Bölümler yüklenemedi. Tekrar deneyin.</p>
                </div>`;
            const button = document.getElementById('isaf-back');
            if (button) button.onclick = () => isaFlixRender();
        }
    }
}


function isaFlixShowEpisodes(series, number, episodes) {
    const root = isaFlixRoot();
    if (!root) return;

    const sortedEpisodes = [...episodes].sort((a, b) => {
        const sa = isaFlixExplicitSeasonNumber(a) || 0;
        const sb = isaFlixExplicitSeasonNumber(b) || 0;
        if (sa !== sb) return sa - sb;

        const an = isaFlixExplicitEpisodeNumber(a);
        const bn = isaFlixExplicitEpisodeNumber(b);
        if (an && bn && an !== bn) return an - bn;
        if (an && !bn) return -1;
        if (!an && bn) return 1;

        return Number(a.__index || 0) - Number(b.__index || 0);
    });

    const seriesTitle = series.title || series.name || 'Dizi';

    root.innerHTML = `
        <div class="isaf-detail">
            <button class="isa-back" id="isaf-season-back">
                <i class="fas fa-arrow-left"></i> Dizilere dön
            </button>
            <h2>${esc(seriesTitle)}</h2>
            <div class="isaf-season-grid">
                ${sortedEpisodes.map((episode, index) => {
                    const label = (index + 1) + '. Bölüm';

                    return `
                        <button class="isaf-season" data-isaf-episode="${index}">
                            <i class="fas fa-circle-play"></i>
                            <strong>${esc(label)}</strong>
                        </button>
                    `;
                }).join('')}
            </div>
        </div>
    `;

    document.getElementById('isaf-season-back').onclick = () => {
        isaFlixState.mode = 'series';
        isaFlixRender();
    };

    root.querySelectorAll('[data-isaf-episode]').forEach(button => {
        button.onclick = () => {
            const episode = sortedEpisodes[Number(button.dataset.isafEpisode)];
            const seriesId = String(series.id || series.series_id || '');

            if (!episode || !episode.id || !seriesId) {
                alert('Bölüm bilgisi bulunamadı.');
                return;
            }

            const seasonNumber =
                isaFlixExplicitSeasonNumber(episode) ||
                (Number(number) > 0 ? Number(number) : 0);
            const episodeNumber = isaFlixExplicitEpisodeNumber(episode);

            const note = document.createElement('p');
            note.className = 'isa-vod-note';
            note.textContent = 'Video hazırlanıyor...';
            root.prepend(note);

            isaFlixNativeCall(
                'memberGetContentPlayback',
                ['episode', String(episode.id), seriesId],
                '__isaContentPlaybackDone',
                result => {
                    note.remove();
                    if (!isaFlixAuthenticated) return;

                    const playbackUrl = String(result?.playback_url || '');
                    if (!result?.ok || !/^https?:\/\//i.test(playbackUrl)) {
                        alert(result?.error || 'Bölüm başlatılamadı.');
                        return;
                    }

                    const showName = cleanMediaTitle(seriesTitle);
                    const entry = {
                        name: showName,
                        url: playbackUrl,
                        logo: episode.image_url || series.poster_url || '',
                        headers: {},
                        season: seasonNumber,
                        episode: episodeNumber,
                        is_live: false,
                        is_series: true,
                        series_id: seriesId,
                        series_name: showName
                    };

                    const queue = sortedEpisodes.map(ep => ({
                        name: showName,
                        url: '',
                        logo: ep.image_url || series.poster_url || '',
                        headers: {},
                        season: isaFlixExplicitSeasonNumber(ep),
                        episode: isaFlixExplicitEpisodeNumber(ep),
                        is_live: false,
                        is_series: true,
                        playback_type: 'episode',
                        item_id: String(ep.id || ''),
                        series_id: seriesId,
                        series_name: showName
                    })).filter(ep => ep.item_id);

                    const selected = queue.findIndex(
                        ep => ep.item_id === String(episode.id)
                    );
                    if (selected >= 0) queue[selected].url = playbackUrl;
                    playIsaFlix(showName, playbackUrl, {}, queue,
                        selected >= 0 ? selected : 0);
                }
            );
        };

    });
}
